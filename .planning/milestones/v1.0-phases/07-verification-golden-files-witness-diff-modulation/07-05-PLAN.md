    Step 1 — Build `tests/rt_safety/hotpath_alloc_gate_target.c`:

    ```c
    /* tests/rt_safety/hotpath_alloc_gate_target.c -- BUILD-06 hot-path target.
     *
     * Initializes SPU-94 with HALL preset; brackets a 10^5-iteration
     * spu94_process hot loop with SIGUSR1 markers; cleans up.
     *
     * Used by hotpath_alloc_gate.sh under strace with a narrow filter
     * (brk, mmap, mmap2, munmap, mremap). Any heap syscall between the
     * two markers fails the gate.
     *
     * Pairs with hotpath_alloc_gate_target_with_malloc.c (negative meta).
     */
    #include <signal.h>
    #include <stdalign.h>
    #include <stdint.h>
    #include <unistd.h>
    #include <spu94/spu94.h>

    #define ITER 100000u
    #define N_SAMPLES 1024u
    #define WORK_BUF_BYTES 524288u  /* 512 KB reverb buffer */

    static alignas(SPU94_STATE_ALIGN_MAX) unsigned char state_buf[SPU94_STATE_SIZE_MAX];
    static alignas(16) unsigned char work_buf[WORK_BUF_BYTES];
    static int16_t Lin[N_SAMPLES], Rin[N_SAMPLES];
    static int16_t Lout[N_SAMPLES], Rout[N_SAMPLES];

    int main(void) {
        spu94_state *s = spu94_init(state_buf, SPU94_STATE_SIZE_MAX,
                                     work_buf, WORK_BUF_BYTES);
        if (!s) return 2;
        spu94_load_preset(s, SPU94_PRESET_HALL);
        /* Trivial input pattern */
        for (unsigned i = 0; i < N_SAMPLES; i++) {
            Lin[i] = Rin[i] = (int16_t)(i & 0x3FFF);
        }

        /* START marker */
        raise(SIGUSR1);

        for (unsigned i = 0; i < ITER; i++) {
            spu94_process(s, Lin, Rin, Lout, Rout, N_SAMPLES);
        }

        /* END marker */
        raise(SIGUSR1);

        spu94_destroy(s);
        return 0;
    }
    ```

    ```bash
    #!/usr/bin/env bash
    # tests/rt_safety/hotpath_alloc_gate.sh -- BUILD-06 D-20 hard gate.
    # Extends the Phase 5 test_no_syscalls pattern with a targeted heap-syscall
    # filter: brk, mmap, mmap2, munmap, mremap (Pitfall 5 covers 32-bit mmap2
    # belt-and-suspenders even though M1 targets are all 64-bit).
    #
    # Exit 0 = zero heap syscalls in steady state.
    # Exit 1 = at least one heap syscall in steady state (CI hard fail).

    set -euo pipefail
    : "${SYSCALLS_BIN:?SYSCALLS_BIN env var required}"
    : "${STRACE_EXE:=strace}"

    LOG=$(mktemp); trap 'rm -f "$LOG"' EXIT

    "$STRACE_EXE" -f -e trace=brk,mmap,mmap2,munmap,mremap \
        -o "$LOG" "$SYSCALLS_BIN" || {
        echo "FAIL: target binary exited non-zero" >&2
        exit 1
    }

    # Find the two SIGUSR1 delivery lines (same marker approach as test_no_syscalls)
    MARKER_LINES=$(grep -n '\-\-\- SIGUSR1' "$LOG" | cut -d: -f1 || true)
    MARKER_COUNT=$(echo "$MARKER_LINES" | grep -c '^[0-9]\+$' || true)
    if [ "$MARKER_COUNT" -ne 2 ]; then
        echo "FAIL: expected 2 SIGUSR1 markers, got $MARKER_COUNT" >&2
        head -20 "$LOG" >&2
        exit 1
    fi

    START=$(echo "$MARKER_LINES" | head -1)
    END=$(echo   "$MARKER_LINES" | tail -1)

    # Count heap syscalls in [START+1, END-1]
    HEAP_HITS=$(sed -n "$((START+1)),$((END-1))p" "$LOG" | \
        grep -cE '^[0-9]+\s+[a-z_]+[0-9]*\((brk|mmap|mmap2|munmap|mremap)' \
          2>/dev/null || true)

    # Fallback matcher (strace without -f, single-process layout):
    if [ -z "$HEAP_HITS" ] || [ "$HEAP_HITS" = "0" ]; then
        HEAP_HITS=$(sed -n "$((START+1)),$((END-1))p" "$LOG" | \
            grep -cE '^(brk|mmap|mmap2|munmap|mremap)\(' 2>/dev/null || true)
    fi

    HEAP_HITS=${HEAP_HITS:-0}

    if [ "$HEAP_HITS" -ne 0 ]; then
        echo "FAIL: $HEAP_HITS heap syscall(s) in spu94_process steady state" >&2
        sed -n "$((START+1)),$((END-1))p" "$LOG" | \
            grep -E '(brk|mmap|mmap2|munmap|mremap)\(' | head -20 >&2
        exit 1
    fi
    echo "PASS: zero heap syscalls in spu94_process hot path"
    exit 0
    ```

    ```python
    import os
    import subprocess
    from pathlib import Path
    import pytest

    GATE = ["bash", "tests/rt_safety/hotpath_alloc_gate.sh"]

    def _find(name: str) -> str:
        # Look under build/ for the ctest-built binary
        hits = list(Path("build").rglob(name))
        assert hits, f"{name} not built under build/"
        return str(hits[0])

    def test_clean_target_passes():
        env = {**os.environ,
               "SYSCALLS_BIN": _find("hotpath_alloc_gate_target"),
               "STRACE_EXE": "strace"}
        r = subprocess.run(GATE, env=env, capture_output=True)
        assert r.returncode == 0, \
            f"clean target unexpectedly failed:\n{r.stderr.decode()}"
        assert b"PASS: zero heap syscalls" in r.stdout

    def test_with_malloc_target_fails():
        env = {**os.environ,
               "SYSCALLS_BIN": _find("hotpath_alloc_gate_target_with_malloc"),
               "STRACE_EXE": "strace"}
        r = subprocess.run(GATE, env=env, capture_output=True)
        assert r.returncode != 0, \
            "with_malloc target unexpectedly passed — gate is broken"
        assert b"FAIL" in r.stderr
        assert any(s in r.stderr for s in (b"brk", b"mmap"))
    ```

    ```python
    """BUILD-06 timing track — pytest-benchmark over spu94_process.

    D-20: timing is REPORT-ONLY; never gates. The only gate is the allocation
    gate in tests/rt_safety/hotpath_alloc_gate.sh (hard fail).
    D-21: tests/benchmarks/benchmark_baselines.json is committed + human-reviewed;
    refreshed only by explicit human action, not by CI.
    """
    import numpy as np
    import pytest
    from spu94 import SPU94, Preset

    PRESET_NAMES = ["off", "room", "studio_a", "studio_b", "studio_c",
                    "hall", "half_echo", "space_echo", "echo", "delay"]
    BLOCK_SIZES = [1024, 4096]
    SEED = 42

    @pytest.fixture(scope="module")
    def input_block_1024():
        rng = np.random.default_rng(SEED)
        return rng.integers(-16000, 16000, size=1024, dtype=np.int16)

    @pytest.fixture(scope="module")
    def input_block_4096():
        rng = np.random.default_rng(SEED)
        return rng.integers(-16000, 16000, size=4096, dtype=np.int16)

    @pytest.mark.benchmark(group="process_1024", min_rounds=5, warmup=True,
                           disable_gc=True)
    @pytest.mark.parametrize("preset_name", PRESET_NAMES)
    def test_bench_process_1024(benchmark, preset_name, input_block_1024):
        s = SPU94()
        s.load_preset(getattr(Preset, preset_name.upper()))
        Lout = np.zeros(1024, dtype=np.int16)
        Rout = np.zeros(1024, dtype=np.int16)
        benchmark(s.process, input_block_1024, input_block_1024, Lout, Rout)

    @pytest.mark.benchmark(group="process_4096", min_rounds=5, warmup=True,
                           disable_gc=True)
    @pytest.mark.parametrize("preset_name", PRESET_NAMES)
    def test_bench_process_4096(benchmark, preset_name, input_block_4096):
        s = SPU94()
        s.load_preset(getattr(Preset, preset_name.upper()))
        Lout = np.zeros(4096, dtype=np.int16)
        Rout = np.zeros(4096, dtype=np.int16)
        benchmark(s.process, input_block_4096, input_block_4096, Lout, Rout)
    ```
