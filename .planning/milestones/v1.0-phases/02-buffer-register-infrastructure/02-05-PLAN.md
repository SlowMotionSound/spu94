**Create `tests/unit/registers/test_register_roundtrip.c`.** Pattern:

```c
/* tests/unit/registers/test_register_roundtrip.c
 * TEST-02: per-register round-trip for all 35 registers. Follows the
 * Phase 1 D-10 pattern: inline hand-constructed reference table with
 * {register, typed-write-value, expected-typed-read-value, policy}. */
#include "unity.h"
#include <spu94/spu94.h>
#include <spu94/spu94_registers.h>
#include <spu94/spu94_register_facade.h>
#include <stdint.h>
#include <stddef.h>
#include <stdalign.h>

void setUp(void) {}
void tearDown(void) {}

/* Reusable state buffer across tests (state-scope, not per-test). */
static alignas(SPU94_STATE_ALIGN_MAX) unsigned char g_state_buf[SPU94_STATE_SIZE_MAX];
static alignas(SPU94_STATE_ALIGN_MAX) unsigned char g_work_buf[1024];
static spu94_state *g_state = (spu94_state *)0;

static void setup_state(void) {
    g_state = spu94_init(g_state_buf, SPU94_STATE_SIZE_MAX, g_work_buf, sizeof(g_work_buf));
    TEST_ASSERT_NOT_NULL(g_state);
}

/* Classification: which registers are i16? u16? */
static int reg_is_u16(spu94_reg_t r) {
    switch (r) {
        case SPU94_REG_mBASE: case SPU94_REG_dAPF1: case SPU94_REG_dAPF2:
        case SPU94_REG_mLSAME: case SPU94_REG_mRSAME:
        case SPU94_REG_mLCOMB1: case SPU94_REG_mRCOMB1:
        case SPU94_REG_mLCOMB2: case SPU94_REG_mRCOMB2:
        case SPU94_REG_dLSAME: case SPU94_REG_dRSAME:
        case SPU94_REG_mLDIFF: case SPU94_REG_mRDIFF:
        case SPU94_REG_mLCOMB3: case SPU94_REG_mRCOMB3:
        case SPU94_REG_mLCOMB4: case SPU94_REG_mRCOMB4:
        case SPU94_REG_dLDIFF: case SPU94_REG_dRDIFF:
        case SPU94_REG_mLAPF1: case SPU94_REG_mRAPF1:
        case SPU94_REG_mLAPF2: case SPU94_REG_mRAPF2:
            return 1;
        default:
            return 0;
    }
}

static int reg_is_immediate(spu94_reg_t r) {
    /* All v* + mBASE = IMMEDIATE. Everything else = TICK_LATCHED. */
    if (r == SPU94_REG_mBASE) return 1;
    return !reg_is_u16(r);  /* i16 (v*) => IMMEDIATE. */
}

void test_every_register_roundtrips(void) {
    setup_state();
    /* Per-register write + tick + read. Use a different test_value per reg
     * so mis-routing (wrong reg written to) fails the comparison. */
    for (int i = 0; i < (int)SPU94_REG__COUNT; ++i) {
        spu94_reg_t r = (spu94_reg_t)i;
        uint16_t test_u = (uint16_t)(0x1000u + (uint16_t)i * 0x111u);  /* distinct per reg, ≤ 0xFFFF */
        int16_t  test_i = (int16_t)(test_u);
        spu94_result_t res;
        if (reg_is_u16(r)) {
            res = spu94_set_reg_u16(g_state, r, test_u);
            TEST_ASSERT_EQUAL_INT(SPU94_OK, res);
        } else {
            res = spu94_set_reg_i16(g_state, r, test_i);
            TEST_ASSERT_EQUAL_INT(SPU94_OK, res);
        }
    }
    /* Tick once to flush pending writes. */
    spu94_tick(g_state);
    /* Now read-back every register and compare. */
    for (int i = 0; i < (int)SPU94_REG__COUNT; ++i) {
        spu94_reg_t r = (spu94_reg_t)i;
        uint16_t test_u = (uint16_t)(0x1000u + (uint16_t)i * 0x111u);
        int16_t  test_i = (int16_t)(test_u);
        char msg[80];
        snprintf(msg, sizeof(msg), "reg %d (%s)", i, spu94_reg_name(r));
        if (reg_is_u16(r)) {
            uint16_t got = spu94_get_reg_u16(g_state, r);
            TEST_ASSERT_EQUAL_UINT16_MESSAGE(test_u, got, msg);
        } else {
            int16_t got = spu94_get_reg_i16(g_state, r);
            TEST_ASSERT_EQUAL_INT16_MESSAGE(test_i, got, msg);
        }
    }
}

void test_snapshot_matches_writes(void) {
    setup_state();
    /* Write every reg with a distinct value, tick, snapshot, compare. */
    for (int i = 0; i < (int)SPU94_REG__COUNT; ++i) {
        spu94_reg_t r = (spu94_reg_t)i;
        uint16_t v = (uint16_t)(0x2000u + (uint16_t)i);
        if (reg_is_u16(r)) spu94_set_reg_u16(g_state, r, v);
        else               spu94_set_reg_i16(g_state, r, (int16_t)v);
    }
    spu94_tick(g_state);
    int16_t out[SPU94_REG__COUNT];
    spu94_snapshot_registers(g_state, out);
    for (int i = 0; i < (int)SPU94_REG__COUNT; ++i) {
        int16_t expected = (int16_t)(0x2000u + (uint16_t)i);
        char msg[80];
        snprintf(msg, sizeof(msg), "reg %d (%s) snapshot", i, spu94_reg_name((spu94_reg_t)i));
        TEST_ASSERT_EQUAL_INT16_MESSAGE(expected, out[i], msg);
    }
}

/* Spot-check facade wrapper parity with engine layer: write via facade,
 * read via engine; confirm identical behavior. */
void test_facade_wrapper_parity(void) {
    setup_state();
    spu94_set_vIIR(g_state, -31415);
    TEST_ASSERT_EQUAL_INT16(-31415, spu94_get_reg_i16(g_state, SPU94_REG_vIIR));
    TEST_ASSERT_EQUAL_INT16(-31415, spu94_get_vIIR(g_state));

    spu94_set_dAPF1(g_state, 0xCAFE);
    spu94_tick(g_state);
    TEST_ASSERT_EQUAL_UINT16(0xCAFE, spu94_get_reg_u16(g_state, SPU94_REG_dAPF1));
    TEST_ASSERT_EQUAL_UINT16(0xCAFE, spu94_get_dAPF1(g_state));
}

int main(void) {
    UNITY_BEGIN();
    RUN_TEST(test_every_register_roundtrips);
    RUN_TEST(test_snapshot_matches_writes);
    RUN_TEST(test_facade_wrapper_parity);
    return UNITY_END();
}
```

```python
def aligned_buffer(size: int, align: int):
    """Allocate a size-byte buffer aligned to `align` bytes."""
    raw = (ctypes.c_ubyte * (size + align))()
    addr = ctypes.addressof(raw)
    offset = (align - (addr % align)) % align
    aligned_ptr = ctypes.c_void_p(addr + offset)
    return raw, aligned_ptr  # keep `raw` alive

    if not state:
        sys.exit("FAIL: spu94_init returned NULL")

    op_log = []  # ring buffer of last 10 ops for failure diagnostics

    for i in range(steps):
        op = rng.randint(0, 2)
        if op == 0:
            # Write mBASE to a random u16 (IMMEDIATE + snap).
            value = rng.randint(0, 0xFFFF)
            lib.spu94_set_reg_u16(state, SPU94_REG_mBASE, value)
            op_desc = f"set_mBASE({value:#x})"
        elif op == 1:
            # Write a random u16 tick-latched register.
            reg = rng.choice(U16_TICK_LATCHED_REGS)
            value = rng.randint(0, 0xFFFF)
            lib.spu94_set_reg_u16(state, reg, value)
            op_desc = f"set_reg_u16(reg={reg}, {value:#x})"
        else:
            lib.spu94_tick(state)
            op_desc = "tick"

        op_log.append(op_desc)
        if len(op_log) > 10:
            op_log.pop(0)

        # Invariant check.
        ba = lib.spu94_get_buffer_address(state)
        mb = lib.spu94_get_reg_u16(state, SPU94_REG_mBASE)
        failures = []
        if ba < mb:
            failures.append(f"buffer_address ({ba:#x}) < mBASE ({mb:#x})")
        if ba > 0x7FFFE:
            failures.append(f"buffer_address ({ba:#x}) > 0x7FFFE")
        if (ba & 1) != 0:
            # Exception: IMMEDIATELY after an odd mBASE write, the snap
            # produces an odd buffer_address. The NEXT tick re-aligns it.
            # We accept this narrow case: if the last op was a set_mBASE
            # with an odd value, ba & 1 may be 1 until the next tick.
            if not (op == 0 and (value & 1) == 1):
                failures.append(f"buffer_address ({ba:#x}) is odd (bit 0 set)")

        if failures:
            print(f"FAIL at step {i}: {failures}", file=sys.stderr)
            print("Last ops:", file=sys.stderr)
            for op in op_log:
                print(f"  {op}", file=sys.stderr)
            print(f"Seed: {seed}", file=sys.stderr)
            return 1

    print(f"OK: {steps} steps passed (seed={seed})")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--seed", type=int, default=0xC0FFEE,
                        help="Random seed (default: 0xC0FFEE; printed on every run).")
    parser.add_argument("--steps", type=int, default=1_000_000,
                        help="Number of fuzz iterations (default: 10^6).")
    parser.add_argument("--lib", type=Path, default=None,
                        help=f"Path to libspu94.so (default: $SPU94_LIB or {DEFAULT_LIB_PATH})")
    args = parser.parse_args()

    lib_path = args.lib
    if lib_path is None:
        env_lib = os.environ.get("SPU94_LIB")
        lib_path = Path(env_lib) if env_lib else DEFAULT_LIB_PATH

    lib = load_lib(lib_path)
    return run_fuzz(lib, args.seed, args.steps)


if __name__ == "__main__":
    sys.exit(main())
```
