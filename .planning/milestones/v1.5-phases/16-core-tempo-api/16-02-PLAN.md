spu94_result_t spu94_set_subdivision(spu94_state *state,
                                     spu94_tempo_reg_t reg,
                                     spu94_subdivision_t subdivision)
{
    if (state == NULL) return SPU94_INVALID_STATE;
    if (state->tempo_bpm == 0) return SPU94_INVALID_ARG;
    if ((int)reg < 0 || (int)reg >= SPU94_TEMPO_REG__COUNT) return SPU94_INVALID_ARG;
    if ((int)subdivision < 0 || (int)subdivision >= SPU94_SUBDIVISION__COUNT) return SPU94_INVALID_ARG;

    const spu94_subdivision_ratio_t *ratio = &spu94_subdivision_table[subdivision];
    uint32_t samples = spu94_compute_delay_samples(state->tempo_bpm,
                                                   ratio->numerator,
                                                   ratio->denominator);
    if (samples > UINT16_MAX) return SPU94_INVALID_ARG;

    /* Write the computed sample count to the hardware register */
    spu94_tempo_writing = 1;
    if ((int)reg < 6) {
        /* d-prefix register: direct write */
        (void)spu94_set_reg_u16(state, tempo_d_prefix_hw_map[reg], (uint16_t)samples);
    } else {
        /* Virtual comb (6-9): compute mCOMB offsets from reference.
         * Combs 0-1 (dCOMB1-2) reference mLSAME/mRSAME.
         * Combs 2-3 (dCOMB3-4) reference mLDIFF/mRDIFF.  */
        int comb_idx = (int)reg - 6;  /* 0-3 */
        uint16_t ref_L, ref_R;
        spu94_reg_t mL_reg, mR_reg;
        if (comb_idx < 2) {
            ref_L = spu94_get_reg_u16(state, SPU94_REG_mLSAME);
            ref_R = spu94_get_reg_u16(state, SPU94_REG_mRSAME);
            mL_reg = (comb_idx == 0) ? SPU94_REG_mLCOMB1 : SPU94_REG_mLCOMB2;
            mR_reg = (comb_idx == 0) ? SPU94_REG_mRCOMB1 : SPU94_REG_mRCOMB2;
        } else {
            ref_L = spu94_get_reg_u16(state, SPU94_REG_mLDIFF);
            ref_R = spu94_get_reg_u16(state, SPU94_REG_mRDIFF);
            mL_reg = (comb_idx == 2) ? SPU94_REG_mLCOMB3 : SPU94_REG_mLCOMB4;
            mR_reg = (comb_idx == 2) ? SPU94_REG_mRCOMB3 : SPU94_REG_mRCOMB4;
        }
        /* Validity: delay must not exceed reference offset */
        if ((uint16_t)samples > ref_L || (uint16_t)samples > ref_R) {
            spu94_tempo_writing = 0;
            return SPU94_INVALID_ARG;
        }
        (void)spu94_set_reg_u16(state, mL_reg, ref_L - (uint16_t)samples);
        (void)spu94_set_reg_u16(state, mR_reg, ref_R - (uint16_t)samples);
    }
    spu94_tempo_writing = 0;

    /* Update binding state to grid-bound (D-07) */
    state->tempo_bind_state[reg] = SPU94_BIND_GRID;
    state->tempo_bind_sub[reg] = (uint8_t)subdivision;
    return SPU94_OK;
}

spu94_result_t spu94_set_tempo(spu94_state *state, uint16_t bpm)
{
    if (state == NULL) return SPU94_INVALID_STATE;
    if (bpm == 0) return SPU94_INVALID_ARG;

    uint16_t old_bpm = state->tempo_bpm;
    state->tempo_bpm = bpm;

    if (old_bpm == 0 || old_bpm == bpm) return SPU94_OK;

    /* Auto-resnap grid-bound registers in active sync groups */
    spu94_tempo_writing = 1;
    for (int i = 0; i < SPU94_TEMPO_REG__COUNT; i++) {
        if (state->tempo_bind_state[i] != SPU94_BIND_GRID) continue;

        /* Check sync group membership */
        if (i < 6 && !state->reflection_sync) continue;
        if (i >= 6 && !state->comb_sync) continue;

        uint8_t sub_idx = state->tempo_bind_sub[i];
        if (sub_idx >= SPU94_SUBDIVISION__COUNT) continue;

        const spu94_subdivision_ratio_t *ratio = &spu94_subdivision_table[sub_idx];
        uint32_t samples = spu94_compute_delay_samples(bpm, ratio->numerator, ratio->denominator);

        if (samples > UINT16_MAX) {
            /* Overflow at new BPM: transition to FIXED (preserves current value) */
            state->tempo_bind_state[i] = SPU94_BIND_FIXED;
            state->tempo_bind_sub[i] = 0xFF;
            continue;
        }

        /* Write the resnapped value */
        if (i < 6) {
            (void)spu94_set_reg_u16(state, tempo_d_prefix_hw_map[i], (uint16_t)samples);
        } else {
            /* Virtual comb resnap */
            int comb_idx = i - 6;
            uint16_t ref_L, ref_R;
            spu94_reg_t mL_reg, mR_reg;
            if (comb_idx < 2) {
                ref_L = spu94_get_reg_u16(state, SPU94_REG_mLSAME);
                ref_R = spu94_get_reg_u16(state, SPU94_REG_mRSAME);
                mL_reg = (comb_idx == 0) ? SPU94_REG_mLCOMB1 : SPU94_REG_mLCOMB2;
                mR_reg = (comb_idx == 0) ? SPU94_REG_mRCOMB1 : SPU94_REG_mRCOMB2;
            } else {
                ref_L = spu94_get_reg_u16(state, SPU94_REG_mLDIFF);
                ref_R = spu94_get_reg_u16(state, SPU94_REG_mRDIFF);
                mL_reg = (comb_idx == 2) ? SPU94_REG_mLCOMB3 : SPU94_REG_mLCOMB4;
                mR_reg = (comb_idx == 2) ? SPU94_REG_mRCOMB3 : SPU94_REG_mRCOMB4;
            }
            if ((uint16_t)samples > ref_L || (uint16_t)samples > ref_R) {
                state->tempo_bind_state[i] = SPU94_BIND_FIXED;
                state->tempo_bind_sub[i] = 0xFF;
                continue;
            }
            (void)spu94_set_reg_u16(state, mL_reg, ref_L - (uint16_t)samples);
            (void)spu94_set_reg_u16(state, mR_reg, ref_R - (uint16_t)samples);
        }
    }
    spu94_tempo_writing = 0;

    /* Proportional register scaling: new_value = old_value * old_bpm / new_bpm */

/* Called by spu94_set_reg_u16 when a tracked d-prefix register is written.
 * Transitions grid-bound registers to proportional state (D-06).
 * Returns immediately if we're inside a tempo snap operation (re-entrancy guard). */
void spu94_tempo_on_reg_write(spu94_state *state, spu94_reg_t reg)
{
    if (spu94_tempo_writing) return;  /* our own write -- don't transition */
    int idx = hw_reg_to_tempo_idx(reg);
    if (idx < 0) return;  /* not a tempo-tracked register */
    if (state->tempo_bind_state[idx] == SPU94_BIND_GRID) {
        state->tempo_bind_state[idx] = SPU94_BIND_PROPORTIONAL;
        state->tempo_bind_ref_bpm[idx] = state->tempo_bpm;
    }
}

/* tests/unit/tempo/test_tempo_binding.c
 * Phase 16 Plan 02 Task 2: Binding state transition tests.
 * Covers D-04 (persistent binding / auto-resnap), D-05 (three states),
 * D-06 (manual write -> proportional), D-07 (re-bind -> grid).
 */

#include "unity.h"
#include <spu94/spu94.h>
#include <spu94/spu94_registers.h>
#include <stdalign.h>
#include <stdint.h>
#include <stddef.h>

/* COVERAGE MAP -- Binding State (D-04..D-07)
 *   D-04: BPM change resnaps grid-bound register   : test_auto_resnap_on_bpm_change
 *   D-04: Resnap only in active sync group         : test_resnap_respects_sync_group
 *   D-05: Initial state is FIXED                   : test_initial_state_fixed
 *   D-05: set_subdivision -> GRID                  : test_bind_to_grid
 *   D-06: Manual write to grid-bound -> PROPORTIONAL : test_manual_write_transitions_proportional
 *   D-06: Manual write to fixed stays fixed        : test_manual_write_fixed_stays_fixed
 *   D-07: Re-bind proportional -> GRID             : test_rebind_proportional_to_grid
 *   D-07: Re-bind fixed -> GRID                    : test_rebind_fixed_to_grid
 *   set_binding_fixed transitions to FIXED         : test_explicit_fixed_transition
 *   Overflow on resnap transitions to FIXED        : test_overflow_resnap_to_fixed
 */

void setUp(void) {}
void tearDown(void) {}

static alignas(SPU94_STATE_ALIGN_MAX) unsigned char g_state_buf[SPU94_STATE_SIZE_MAX];
static unsigned char g_work_buf[SPU94_WORK_BUF_MAX_BYTES];

static spu94_state *fresh_state(void) {
    return spu94_init(g_state_buf, sizeof(g_state_buf), g_work_buf, sizeof(g_work_buf));
}

void test_initial_state_fixed(void) {
    spu94_state *s = fresh_state();
    TEST_ASSERT_NOT_NULL(s);
    for (int i = 0; i < SPU94_TEMPO_REG__COUNT; i++) {

    spu94_set_subdivision(s, SPU94_TEMPO_REG_dAPF1, SPU94_SUB_1_4);
    TEST_ASSERT_EQUAL_INT(SPU94_BIND_GRID,
        spu94_get_binding_state(s, SPU94_TEMPO_REG_dAPF1));

void test_auto_resnap_on_bpm_change(void) {
    spu94_state *s = fresh_state();
    spu94_set_tempo(s, 120);
    spu94_set_reflection_sync(s, 1);
    spu94_set_subdivision(s, SPU94_TEMPO_REG_dAPF1, SPU94_SUB_1_4);

    /* At 120 BPM, 1/4: (60*22050*1)/(120*4) = 2756 */
    spu94_tick(s);  /* commit pending */
    uint16_t val_120 = spu94_get_reg_u16(s, SPU94_REG_dAPF1);

    /* Change to 60 BPM: (60*22050*1)/(60*4) = 5512 */
    spu94_set_tempo(s, 60);
    spu94_tick(s);
    uint16_t val_60 = spu94_get_reg_u16(s, SPU94_REG_dAPF1);

    TEST_ASSERT_EQUAL_UINT16(2756, val_120);
    TEST_ASSERT_EQUAL_UINT16(5512, val_60);
    /* Still grid-bound */
    TEST_ASSERT_EQUAL_INT(SPU94_BIND_GRID,
        spu94_get_binding_state(s, SPU94_TEMPO_REG_dAPF1));
}
