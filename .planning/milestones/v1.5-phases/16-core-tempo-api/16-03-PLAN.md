Comb grouping:
- dCOMB1 (idx 6): writes mLCOMB1, mRCOMB1. Reference = mLSAME, mRSAME.
- dCOMB2 (idx 7): writes mLCOMB2, mRCOMB2. Reference = mLSAME, mRSAME.

Computation: mCOMB = reference_offset - delay_samples

- mLDIFF = 0x11BC (4540), mRDIFF = 0x1410 (5136)
- mLCOMB1 = 0x14C2 (5314), mLCOMB2 = 0x12C2 (4802)
- mLCOMB3 = 0x0C02 (3074), mLCOMB4 = 0x09C2 (2498)

/* tests/unit/tempo/test_tempo_comb.c
 * Phase 16 Plan 03 Task 1: Virtual comb-delay computation tests.
 * Verifies dCOMB1-4 correctly compute mCOMB addresses from desired
 * delay time relative to buffer geometry (mLSAME/mRSAME or mLDIFF/mRDIFF).
 */

#include "unity.h"
#include <spu94/spu94.h>
#include <spu94/spu94_registers.h>
#include <stdalign.h>
#include <stdint.h>
#include <stddef.h>

/* COVERAGE MAP -- Virtual Comb Delays
 *   dCOMB1 writes mLCOMB1/mRCOMB1 correctly  : test_comb1_writes_correct_offsets
 *   dCOMB2 writes mLCOMB2/mRCOMB2 correctly  : test_comb2_writes_correct_offsets
 *   dCOMB3 writes mLCOMB3/mRCOMB3 from DIFF  : test_comb3_writes_from_diff_reference
 *   dCOMB4 writes mLCOMB4/mRCOMB4 from DIFF  : test_comb4_writes_from_diff_reference
 *   Comb delay > geometry rejected            : test_comb_delay_exceeds_geometry
 *   Comb sync off: no resnap                  : test_comb_sync_off_no_resnap
 *   Comb sync on: resnap on BPM change        : test_comb_sync_on_resnaps
 *   All 15 subs valid for combs at standard BPMs : test_comb_all_subs_at_120
 */

void setUp(void) {}
void tearDown(void) {}

static alignas(SPU94_STATE_ALIGN_MAX) unsigned char g_state_buf[SPU94_STATE_SIZE_MAX];
static unsigned char g_work_buf[SPU94_WORK_BUF_MAX_BYTES];

static spu94_state *fresh_state_with_hall(void) {
    spu94_state *s = spu94_init(g_state_buf, sizeof(g_state_buf),
                                g_work_buf, sizeof(g_work_buf));
    if (s) {
        spu94_load_preset(s, SPU94_PRESET_HALL);
        spu94_tick(s);  /* commit TICK_LATCHED registers */
    }
    return s;
}

void test_comb1_writes_correct_offsets(void) {
    spu94_state *s = fresh_state_with_hall();
    TEST_ASSERT_NOT_NULL(s);
    spu94_set_tempo(s, 120);

    /* Hall: mLSAME=5562, mRSAME=6082 */
    uint16_t mLSAME = spu94_get_reg_u16(s, SPU94_REG_mLSAME);
    uint16_t mRSAME = spu94_get_reg_u16(s, SPU94_REG_mRSAME);

    /* 120 BPM, 1/16: (60*22050*1)/(120*16) = 689 samples delay */
    TEST_ASSERT_EQUAL_INT(SPU94_OK,
        spu94_set_subdivision(s, SPU94_TEMPO_REG_dCOMB1, SPU94_SUB_1_16));

    spu94_tick(s);
    /* mLCOMB1 = mLSAME - 689 */
    TEST_ASSERT_EQUAL_UINT16(mLSAME - 689,
        spu94_get_reg_u16(s, SPU94_REG_mLCOMB1));
    /* mRCOMB1 = mRSAME - 689 */
    TEST_ASSERT_EQUAL_UINT16(mRSAME - 689,
        spu94_get_reg_u16(s, SPU94_REG_mRCOMB1));
}

void test_comb_delay_exceeds_geometry(void) {
    spu94_state *s = fresh_state_with_hall();
    TEST_ASSERT_NOT_NULL(s);
    /* Use a very low BPM to generate a large delay */
    spu94_set_tempo(s, 40);

    /* 40 BPM, 1/1: (60*22050*1)/(40*1) = 33075 samples.
     * Hall mLSAME = 5562 -- delay > geometry. Should be rejected. */
    TEST_ASSERT_EQUAL_INT(SPU94_INVALID_ARG,
        spu94_set_subdivision(s, SPU94_TEMPO_REG_dCOMB1, SPU94_SUB_1_1));
}

void test_comb_sync_off_no_resnap(void) {
    spu94_state *s = fresh_state_with_hall();
    TEST_ASSERT_NOT_NULL(s);
    spu94_set_tempo(s, 120);
    spu94_set_comb_sync(s, 0);  /* OFF */

    spu94_set_subdivision(s, SPU94_TEMPO_REG_dCOMB1, SPU94_SUB_1_16);
    spu94_tick(s);
    uint16_t val_before = spu94_get_reg_u16(s, SPU94_REG_mLCOMB1);

    /* BPM change -- comb sync OFF means no resnap */
    spu94_set_tempo(s, 60);
    spu94_tick(s);
    uint16_t val_after = spu94_get_reg_u16(s, SPU94_REG_mLCOMB1);

    TEST_ASSERT_EQUAL_UINT16(val_before, val_after);
}

void test_comb_sync_on_resnaps(void) {
    spu94_state *s = fresh_state_with_hall();
    TEST_ASSERT_NOT_NULL(s);
    spu94_set_tempo(s, 120);
    spu94_set_comb_sync(s, 1);  /* ON */

    uint16_t mLSAME = spu94_get_reg_u16(s, SPU94_REG_mLSAME);

    spu94_set_subdivision(s, SPU94_TEMPO_REG_dCOMB1, SPU94_SUB_1_16);
    spu94_tick(s);
    /* 120 BPM 1/16: 689 samples -> mLCOMB1 = mLSAME - 689 */
    TEST_ASSERT_EQUAL_UINT16(mLSAME - 689,
        spu94_get_reg_u16(s, SPU94_REG_mLCOMB1));

    /* BPM -> 240: 1/16 = (60*22050*1)/(240*16) = 344 samples */
    spu94_set_tempo(s, 240);
    spu94_tick(s);
    TEST_ASSERT_EQUAL_UINT16(mLSAME - 344,
        spu94_get_reg_u16(s, SPU94_REG_mLCOMB1));
}

void test_comb_all_subs_at_120(void) {
    spu94_state *s = fresh_state_with_hall();
    TEST_ASSERT_NOT_NULL(s);
    spu94_set_tempo(s, 120);

    uint16_t mLSAME = spu94_get_reg_u16(s, SPU94_REG_mLSAME);

    /* All 15 subdivisions should either succeed (delay < mLSAME)
     * or fail cleanly with INVALID_ARG (delay > geometry).
     * At 120 BPM, the longest is 1/1 dotted = (60*22050*3)/(120*2) = 16537.
     * Hall mLSAME = 5562 -> only subs producing delay <= 5562 will succeed. */
    int valid_count = 0;
    for (int i = 0; i < SPU94_SUBDIVISION__COUNT; i++) {
        spu94_result_t rc = spu94_set_subdivision(s, SPU94_TEMPO_REG_dCOMB1,
                                                  (spu94_subdivision_t)i);
        if (rc == SPU94_OK) valid_count++;
        else TEST_ASSERT_EQUAL_INT(SPU94_INVALID_ARG, rc);
    }
    /* At 120 BPM with Hall geometry, short subdivisions (1/8, 1/16, etc.) should pass */
    TEST_ASSERT_GREATER_THAN(0, valid_count);
    /* 1/16 = 689, 1/16d = 1033, 1/16t = 459, 1/8 = 1378, 1/8d = 2067,
     * 1/8t = 918, 1/4 = 2756, 1/4d = 4134, 1/4t = 1837 -- all < 5562
     * 1/2 = 5512 < 5562 ok. 1/2d = 8268 > 5562 fail. 1/2t = 3675 ok.
     * 1/1 = 11025 fail. 1/1d = 16537 fail. 1/1t = 7350 fail. */
    /* Expected: 11 valid, 4 rejected */
    TEST_ASSERT_EQUAL_INT(11, valid_count);
}

int main(void) {
    UNITY_BEGIN();
    RUN_TEST(test_comb1_writes_correct_offsets);
    RUN_TEST(test_comb2_writes_correct_offsets);
    RUN_TEST(test_comb3_writes_from_diff_reference);
    RUN_TEST(test_comb4_writes_from_diff_reference);
    RUN_TEST(test_comb_delay_exceeds_geometry);
    RUN_TEST(test_comb_sync_off_no_resnap);
    RUN_TEST(test_comb_sync_on_resnaps);
    RUN_TEST(test_comb_all_subs_at_120);
    return UNITY_END();
}
