```cpp
// Tempo sync state push (Phase 18, TEMPO-08/TEMPO-09)
{
    uint8_t mode = syncMode.load(std::memory_order_relaxed);
    uint8_t gSub = globalSubdivision.load(std::memory_order_relaxed);

    if (mode != 0) {  // INT or EXT
        uint16_t bpm = tempoBpm.load(std::memory_order_relaxed);

        // On mode transition from FREE: enable sync groups and set up bindings
        if (lastPushedMode == 0 && mode != 0) {
            spu94_set_reflection_sync(spu, 1);
            spu94_set_comb_sync(spu, 1);

            // Apply all per-register subdivisions
            for (int r = 0; r < SPU94_TEMPO_REG__COUNT; r++) {
                uint8_t regSub = perRegSub[r].load(std::memory_order_relaxed);
                if (regSub == SPU94_SUBDIVISION__COUNT) {
                    // "Free" -- set register to FIXED (unclocked)
                    spu94_set_binding_fixed(spu, (spu94_tempo_reg_t)r);
                } else {
                    uint8_t actualSub = (regSub == 0xFF) ? gSub : regSub;
                    spu94_set_subdivision(spu, (spu94_tempo_reg_t)r,
                                          (spu94_subdivision_t)actualSub);
                }
                lastPushedSub[r] = perRegSub[r].load(std::memory_order_relaxed);
            }
            lastPushedGlobalSub = gSub;

            if (bpm > 0) {
                spu94_set_tempo(spu, bpm);
                lastPushedBpm = bpm;
            }
        } else {
            // Steady-state: check for individual changes

            // BPM change
            if (bpm > 0 && bpm != lastPushedBpm) {
                spu94_set_tempo(spu, bpm);
                lastPushedBpm = bpm;
            }

            // Global subdivision change: resnap all "Global" registers
            if (gSub != lastPushedGlobalSub) {
                for (int r = 0; r < SPU94_TEMPO_REG__COUNT; r++) {
                    uint8_t regSub = perRegSub[r].load(std::memory_order_relaxed);
                    if (regSub == 0xFF) {  // "Global"
                        spu94_set_subdivision(spu, (spu94_tempo_reg_t)r,
                                              (spu94_subdivision_t)gSub);
                    }
                }
                lastPushedGlobalSub = gSub;
            }

            // Per-register subdivision changes
            for (int r = 0; r < SPU94_TEMPO_REG__COUNT; r++) {
                uint8_t regSub = perRegSub[r].load(std::memory_order_relaxed);
                if (regSub != lastPushedSub[r]) {
                    if (regSub == SPU94_SUBDIVISION__COUNT) {
                        spu94_set_binding_fixed(spu, (spu94_tempo_reg_t)r);
                    } else {
                        uint8_t actualSub = (regSub == 0xFF) ? gSub : regSub;
                        spu94_set_subdivision(spu, (spu94_tempo_reg_t)r,
                                              (spu94_subdivision_t)actualSub);
                    }
                    lastPushedSub[r] = regSub;
                }
            }
        }
    } else if (lastPushedMode != 0) {
        // Transition from INT/EXT to FREE: disable sync
        // Registers keep their current sample values (no resnap)
        lastPushedBpm = 0;
    }
    lastPushedMode = mode;

    // Update binding state shadows for GUI consumption
    for (int r = 0; r < SPU94_TEMPO_REG__COUNT; r++) {
        bindingStateShadow[r].store(
            (uint8_t)spu94_get_binding_state(spu, (spu94_tempo_reg_t)r),
            std::memory_order_relaxed);
        bindingSubShadow[r].store(
            (uint8_t)spu94_get_binding_subdivision(spu, (spu94_tempo_reg_t)r),
            std::memory_order_relaxed);
    }
}
```
