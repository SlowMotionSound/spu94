4. Add tempo application AFTER the `spu94_tick(state)` call (line ~420) and AFTER factory preset fader setup (line ~431), but BEFORE user fader overrides. This implements D-05 (--tempo applied after preset loading). Insert this block:
   ```c
   /* Tempo sync (TEMPO-07, D-04/D-05): applied after preset/config loading
    * and the initial tick so register values are committed.
    * - Factory presets have no tempo state: set default 1/4 subdivision
    *   on all registers that are still FIXED.
    * - File presets may carry tempo state: existing bindings are preserved,
    *   spu94_set_tempo resnaps all GRID-bound registers to the new BPM. */
   if (tempo_bpm > 0) {
       spu94_set_reflection_sync(state, 1);
       spu94_set_comb_sync(state, 1);

       /* For factory presets (no tempo state in the preset): assign default
        * 1/4 note subdivision on all registers that are still FIXED. */
       if (preset_name) {
           for (int r = 0; r < SPU94_TEMPO_REG__COUNT; r++) {
               if (spu94_get_binding_state(state, (spu94_tempo_reg_t)r) == SPU94_BIND_FIXED) {
                   spu94_set_subdivision(state, (spu94_tempo_reg_t)r, SPU94_SUB_1_4);
               }
           }
       }

       spu94_result_t trc = spu94_set_tempo(state, tempo_bpm);
       if (trc != SPU94_OK) {
           SPU94_ERROR("failed to set tempo to %u BPM", (unsigned)tempo_bpm);
           spu94_destroy(state);
           free(input.L); free(input.R); free(work_buf);
           return 2;
       }
       spu94_tick(state);  /* commit resnapped registers */
   }
   ```

import subprocess
from pathlib import Path
