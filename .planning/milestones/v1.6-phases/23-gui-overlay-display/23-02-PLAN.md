**Unit conversion statics:**

```cpp
juce::String MacroPanel::regToMs(int16_t val) {
    float ms = static_cast<float>(val) / 22050.0f * 1000.0f;
    if (ms < 10.0f) return juce::String(ms, 2) + " ms";
    return juce::String(ms, 1) + " ms";
}

juce::String MacroPanel::regToMeters(uint16_t val) {
    float meters = static_cast<float>(val) * 343.0f / 22050.0f;
    if (meters < 1.0f) return juce::String(meters * 100.0f, 1) + " cm";
    return juce::String(meters, 2) + " m";
}

juce::String MacroPanel::regToPercent(int16_t val) {
    float pct = static_cast<float>(val) / 32767.0f * 100.0f;
    return juce::String(pct, 1) + "%";
}

juce::String MacroPanel::regToPercentBipolar(int16_t val, int16_t floor, int16_t ceiling) {
    if (val >= 0) {
        float pct = (ceiling > 0) ? (static_cast<float>(val) / ceiling * 100.0f) : 0.0f;
        return "+" + juce::String(pct, 1) + "%";
    } else {
        float pct = (floor < 0) ? (static_cast<float>(val) / static_cast<float>(-floor) * -100.0f) : 0.0f;
        return juce::String(pct, 1) + "%";
    }
}
```
