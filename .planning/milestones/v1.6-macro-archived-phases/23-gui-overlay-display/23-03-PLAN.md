```cpp
RegisterPanel::RegDisplayType RegisterPanel::classifyRegister(spu94_reg_t reg) {
    // d-prefix registers: delay offsets (dAPF1, dAPF2, dLSAME, dRSAME, dLDIFF, dRDIFF)
    if (reg == SPU94_REG_dAPF1 || reg == SPU94_REG_dAPF2 ||
        reg == SPU94_REG_dLSAME || reg == SPU94_REG_dRSAME ||
        reg == SPU94_REG_dLDIFF || reg == SPU94_REG_dRDIFF)
        return RegDisplayType::DELAY;
    // m-prefix address registers (excluding mBASE)
    if (spu94_reg_type(reg) == SPU94_REG_TYPE_U16 && reg != SPU94_REG_mBASE)
        return RegDisplayType::ADDRESS;
    // mBASE gets its own type
    if (reg == SPU94_REG_mBASE)
        return RegDisplayType::BASE;
    // v-prefix coefficients (vIIR, vWALL, vCOMB1-4, vAPF1-2, vLOUT, vROUT, vLIN, vRIN)
    return RegDisplayType::COEFF;
}
```

**3. Add dual readout textFromValueFunction (UNIT-02):**
In the constructor, within the slider setup loop, AFTER setting up each slider's range and callbacks, add a textFromValueFunction. The dual readout format is "0xABCD / 12.3 ms":

```cpp
for (size_t i = 0; i < kSliderRegisters.size(); ++i) {
    const spu94_reg_t reg = kSliderRegisters[i];
    const RegDisplayType displayType = classifyRegister(reg);

    sliders[i].textFromValueFunction = [displayType](double val) -> juce::String {
        int16_t v = static_cast<int16_t>(val);
        uint16_t uv = static_cast<uint16_t>(v);
        // Hex part
        juce::String hex = "0x" + juce::String::toHexString(uv)
            .toUpperCase().paddedLeft('0', 4);
        // Human part
        juce::String human;
        switch (displayType) {
            case RegDisplayType::DELAY: {
                float ms = static_cast<float>(v) / 22050.0f * 1000.0f;
                human = (ms < 10.0f) ? juce::String(ms, 2) + " ms"
                                     : juce::String(ms, 1) + " ms";
                break;
            }
            case RegDisplayType::ADDRESS: {
                float meters = static_cast<float>(uv) * 343.0f / 22050.0f;
                human = (meters < 1.0f) ? juce::String(meters * 100.0f, 1) + " cm"
                                        : juce::String(meters, 2) + " m";
                break;
            }
            case RegDisplayType::BASE: {
                // mBASE: show as buffer allocation
                human = juce::String(uv) + " hw";  // halfwords
                break;
            }
            case RegDisplayType::COEFF: {
                float pct = static_cast<float>(v) / 32767.0f * 100.0f;
                human = juce::String(pct, 1) + "%";
                break;
            }
        }
        return hex + " / " + human;
    };
}
```

Make sure the textFromValueFunction is added INSIDE the existing slider construction loop (around line 6-40 in RegisterPanel.cpp). The existing loop already iterates `for (size_t i = 0; i < kSliderRegisters.size(); ++i)` and sets up each slider.
