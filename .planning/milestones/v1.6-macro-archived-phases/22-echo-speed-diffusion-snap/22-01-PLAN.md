/* -----------------------------------------------------------------
 * Phase 22 (Echo Speed + Diffusion Snap, SNAP-01..03): Sync/Free
 * modal toggle for echo speed and diffusion texture controls.
 *
 * In Sync mode, registers are quantized to subdivision table positions.
 * Sweep/Spread/Rotate macros operate on table indices (echo speed only).
 * dAPF1/dAPF2 get toggle + dropdown only (no macros per D-12).
 *
 * State budget: 2 + 6 + 12 = 20 bytes. Total struct ~3684B of 16384.
 * ----------------------------------------------------------------- */
uint8_t        echo_speed_sync;         /* 0=Free, 1=Sync (D-01) */
uint8_t        diff_texture_sync;       /* 0=Free, 1=Sync (D-12) */

 * Echo speed: 4 registers (dLSAME, dRSAME, dLDIFF, dRDIFF) per D-03.
 * Diff texture: 2 registers (dAPF1, dAPF2) per D-12.
 * Values: 0-14 (spu94_subdivision_t) or 0xFF (unset/no selection). */
uint8_t        snap_echo_sub[4];        /* dropdown selections */
uint8_t        snap_diff_sub[2];        /* dropdown selections */
