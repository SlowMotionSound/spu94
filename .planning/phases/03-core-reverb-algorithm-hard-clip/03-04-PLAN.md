    static uint32_t lcg(uint32_t *s) { *s = (*s) * 1103515245u + 12345u; return *s; }

        uint32_t seed = 0x12345678u;
        for (size_t i = 0; i < sizeof(gA_work); ++i) { gA_work[i] = (unsigned char)lcg(&seed); }
        memcpy(gB_work, gA_work, sizeof(gB_work));

        spu94_state *A = spu94_init(gA_state, sizeof(gA_state), gA_work, sizeof(gA_work));

        seed = 0x12345678u;  /* reset for reg values */

                spu94_set_reg_i16(A, (spu94_reg_t)r, v);
                spu94_set_reg_i16(B, (spu94_reg_t)r, v);
