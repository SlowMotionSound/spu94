Create `include/spu94/spu94_adpcm.h` with:

```c
#ifndef SPU94_ADPCM_H
#define SPU94_ADPCM_H

#include <stdint.h>

#ifdef __cplusplus
extern "C" {
#endif

/* ADPCM decoder state: two previous decoded samples.
 * Caller allocates and zero-initializes for a new stream.
 * State carries across block boundaries within a stream. */
typedef struct {
    int16_t old;    /* previous decoded sample */
    int16_t older;  /* sample before previous */
} spu94_adpcm_state;

/* 5 fixed SPU ADPCM filter coefficient pairs.
 * f0 (positive): {0, 60, 115, 98, 122}
 * f1 (negative): {0,  0, -52, -55, -60}
 * Defined in spu94_adpcm.c. Exposed for test/diagnostic use. */
extern const int16_t spu94_adpcm_f0[5];
extern const int16_t spu94_adpcm_f1[5];

/* Samples per ADPCM block */
#define SPU94_ADPCM_BLOCK_SAMPLES  28
/* Bytes per ADPCM block */
#define SPU94_ADPCM_BLOCK_BYTES    16

/* Decode one 16-byte ADPCM block into 28 int16_t samples.
 *
 * state: caller-allocated, zero-init for new stream. Updated in-place.
 * block: 16-byte ADPCM block (byte 0 = shift|filter, byte 1 = flags,
 *        bytes 2-15 = 14 data bytes with 2 nibbles each).
 * out:   28 decoded int16_t samples.
 *
 * Returns the flag byte (block[1]) for caller to handle loop logic.
 *
 * Arithmetic: (old*f0 + older*f1 + 32) >> 6 (ASR, per ADR-0001 discipline).
 * Shift 13-15 mapped to 9. Filter 5-7 clamped to 4. Clamp to int16 after
 * full prediction. Nibble order: low nibble first within each data byte. */
uint8_t spu94_adpcm_decode_block(
    spu94_adpcm_state *state,
    const uint8_t      block[SPU94_ADPCM_BLOCK_BYTES],
    int16_t            out[SPU94_ADPCM_BLOCK_SAMPLES]
);

#ifdef __cplusplus
}
#endif

#endif /* SPU94_ADPCM_H */
```

Create `src/spu94/spu94_adpcm.c` with:

1. Include `<spu94/spu94_adpcm.h>` and `<spu94/spu94_q15.h>` (for `sat_s16`).

2. Filter coefficient tables as file-scope const arrays:
```c
const int16_t spu94_adpcm_f0[5] = {  0, 60, 115,  98, 122 };
const int16_t spu94_adpcm_f1[5] = {  0,  0, -52, -55, -60 };
```

3. `spu94_adpcm_decode_block` implementation:
   - Parse header: `shift = block[0] & 0x0F; filter = (block[0] >> 4) & 0x07;`
   - Clamp shift: `if (shift > 12) shift = 9;`
   - Clamp filter: `if (filter > 4) filter = 4;`
   - Look up `f0 = spu94_adpcm_f0[filter]; f1 = spu94_adpcm_f1[filter];`
   - Load state: `int32_t old = state->old; int32_t older = state->older;`
   - Loop i=0..13 (14 data bytes, bytes 2-15):
     - `uint8_t byte = block[2 + i];`
     - For each of 2 nibbles (low first, high second):
       - Extract nibble: low = `byte & 0x0F`, high = `(byte >> 4) & 0x0F`
       - Sign-extend: `if (nibble >= 8) nibble -= 16;` (range -8..+7)
       - Shift: `int32_t shifted = nibble << (12 - shift);`
       - Predict: `int32_t predicted = (old * f0 + older * f1 + 32) >> 6;`  (ASR)
       - Sum: `int32_t sample = shifted + predicted;`
       - Clamp: `int16_t clamped = sat_s16(sample);`
       - Store: `out[i*2 + nibble_index] = clamped;`
       - Update: `older = old; old = (int32_t)clamped;`
   - Write back state: `state->old = (int16_t)old; state->older = (int16_t)older;`
   - Return `block[1]` (flag byte).

4. Add `spu94_adpcm.c` to the `spu94_obj` OBJECT library source list in `src/spu94/CMakeLists.txt` (append after `spu94_presets.c`).

Do NOT use `/64` anywhere — use `>> 6` exclusively per ASR discipline. All intermediate arithmetic in int32_t. No float. No heap.

```c
static void build_block(uint8_t block[16], int shift, int filter, uint8_t flags,
                        const int8_t nibbles[28]) {
    memset(block, 0, 16);
    block[0] = (uint8_t)((filter << 4) | (shift & 0x0F));
    block[1] = flags;
    for (int i = 0; i < 14; i++) {
        uint8_t lo = (uint8_t)(nibbles[i*2]     & 0x0F);
        uint8_t hi = (uint8_t)(nibbles[i*2 + 1] & 0x0F);
        block[2 + i] = lo | (hi << 4);
    }
}
```
