```c
int16_t spu94_dac_noise_step_8x(spu94_dac_noise_state *state) {
    /* Identical LFSR + HP shaping as spu94_dac_noise_step, but uses
     * DAC_NOISE_SHIFT_8X for 352.8kHz amplitude calibration. */
    uint32_t lfsr = state->lfsr;
    uint32_t bit = lfsr & 1u;
    lfsr >>= 1;
    if (bit) lfsr ^= DAC_NOISE_LFSR_FEEDBACK;
    state->lfsr = lfsr;

    int16_t x = (int16_t)(((int32_t)(lfsr >> 16) - 32768) >> DAC_NOISE_SHIFT_8X);

    int32_t y = (int32_t)x - 2 * (int32_t)state->x_prev + (int32_t)state->x_prev2;
    state->x_prev2 = state->x_prev;
    state->x_prev = x;

    return sat_s16(y);
}
```

The function body:
```c
int16_t spu94_dac_fir_step_8x_with_noise(spu94_dac_fir_state *fir,
                                          spu94_dac_noise_state *noise,
                                          int16_t input)
{
    /* Stages 1 and 2 are identical to spu94_dac_fir_step_8x */

    /* Stage 1: 2 evaluations at 88.2kHz */
    int16_t s1[2];
    dac_fir_push(fir->stage1_delay, &fir->stage1_idx,
                 input, DAC_FIR_STAGE1_NTAPS);
    s1[0] = dac_fir_stage_apply(fir->stage1_delay, fir->stage1_idx,
                                DAC_FIR_STAGE1_NTAPS,
                                dac_interp_stage1,
                                dac_fir_stage1_pairs,
                                DAC_FIR_STAGE1_NPAIRS);
    dac_fir_push(fir->stage1_delay, &fir->stage1_idx,
                 0, DAC_FIR_STAGE1_NTAPS);
    s1[1] = dac_fir_stage_apply(fir->stage1_delay, fir->stage1_idx,
                                DAC_FIR_STAGE1_NTAPS,
                                dac_interp_stage1,
                                dac_fir_stage1_pairs,
                                DAC_FIR_STAGE1_NPAIRS);

    /* Stage 2: 4 evaluations at 176.4kHz */
    int16_t s2[4];
    for (int i = 0; i < 2; i++) {
        dac_fir_push(fir->stage2_delay, &fir->stage2_idx,
                     s1[i], DAC_FIR_STAGE2_NTAPS);
        s2[2 * i] = dac_fir_stage_apply(fir->stage2_delay, fir->stage2_idx,
                                        DAC_FIR_STAGE2_NTAPS,
                                        dac_interp_stage2,
                                        dac_fir_stage2_pairs,
                                        DAC_FIR_STAGE2_NPAIRS);
        dac_fir_push(fir->stage2_delay, &fir->stage2_idx,
                     0, DAC_FIR_STAGE2_NTAPS);
        s2[2 * i + 1] = dac_fir_stage_apply(fir->stage2_delay, fir->stage2_idx,
                                            DAC_FIR_STAGE2_NTAPS,
                                            dac_interp_stage2,
                                            dac_fir_stage2_pairs,
                                            DAC_FIR_STAGE2_NPAIRS);
    }

    /* Stage 3: 8 evaluations at 352.8kHz -- NOISE INJECTED AT EACH TICK.
     * Per D-03: noise originates at converter clock rate (352.8kHz) and
     * gets filtered by the reconstruction path, producing spectrally
     * shaped noise rather than flat white noise.
     * Per Pitfall 4: noise is added BEFORE decimation pick, so the <<3
     * gain compensation amplifies both signal and noise. The
     * DAC_NOISE_SHIFT_8X calibration accounts for this 18dB boost. */
    int16_t s3_last = 0;
    for (int j = 0; j < 4; j++) {
        dac_fir_push(fir->stage3_delay, &fir->stage3_idx,
                     s2[j], DAC_FIR_STAGE3_NTAPS);
        int16_t s3_val = dac_fir_stage_apply(fir->stage3_delay, fir->stage3_idx,
                                              DAC_FIR_STAGE3_NTAPS,
                                              dac_interp_stage3,
                                              dac_fir_stage3_pairs,
                                              DAC_FIR_STAGE3_NPAIRS);
        s3_val = q15_add_sat(s3_val, spu94_dac_noise_step_8x(noise));

        dac_fir_push(fir->stage3_delay, &fir->stage3_idx,
                     0, DAC_FIR_STAGE3_NTAPS);
        s3_last = dac_fir_stage_apply(fir->stage3_delay, fir->stage3_idx,
                                      DAC_FIR_STAGE3_NTAPS,
                                      dac_interp_stage3,
                                      dac_fir_stage3_pairs,
                                      DAC_FIR_STAGE3_NPAIRS);
        s3_last = q15_add_sat(s3_last, spu94_dac_noise_step_8x(noise));
    }

    /* Gain compensation + decimation (identical to spu94_dac_fir_step_8x) */
    return sat_s16((int32_t)s3_last << 3);
}
```

IMPORTANT: The `(void)` discard in `spu94_dac_fir_step_8x` for intermediate Stage 3
results is replaced by `s3_val` which receives noise. The intermediate s3_val is NOT
kept for decimation (only s3_last survives), but the noise step MUST still be called
to advance the LFSR state -- all 8 ticks must execute per output sample. The noise
added to s3_val feeds back into nothing (it is discarded), but the LFSR state advance
is the critical side effect.
