```c
static const struct option long_opts[] = {
    {"preset",       required_argument, NULL, 'p'},
    {"config",       required_argument, NULL, 'c'},
    {"tail-seconds", required_argument, NULL, 't'},
    {"list-presets", no_argument,       NULL, 'l'},
    {"help",         no_argument,       NULL, 'h'},
    {NULL, 0, NULL, 0}
};
int opt;
while ((opt = getopt_long(argc, argv, "p:c:t:lh", long_opts, NULL)) != -1) {
    switch (opt) {
        case 'p': preset_name = optarg; break;
        case 'c': config_path = optarg; break;
        case 't': tail_seconds = strtod(optarg, NULL); break;
        case 'l': list_presets_and_exit(); return 0;
        case 'h': print_help(); return 0;
        default:  fprintf(stderr, "spu94: error: try --help\n"); return 2;
    }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| argv → CLI | User-controlled command-line arguments; validated by getopt_long + per-flag range checks. |
| --config JSON file → CLI | User-provided file content; jsmn tokenizes; register-name + value range checks. |
| --input WAV file → dr_wav | User-provided file; dr_wav handles malformed WAV gracefully with error return. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-14 | Tampering | Malformed WAV causes heap/stack corruption in dr_wav | mitigate | Use upstream dr_wav v0.14.6 unchanged; all dr_wav init APIs return 0 on failure — the CLI treats that as the CLI-04 one-line-error path, never passes malformed data further. Error: `spu94: error: could not read WAV file '%s'` |
| T-06-15 | Tampering | Malformed JSON causes jsmn failure or overflow | mitigate | Use jsmn v1.1.0 unchanged. jsmn is zero-allocation; token array is caller-sized (`MAX_TOKENS=1024`). `jsmn_parse()` returns negative on failure (JSMN_ERROR_INVAL / JSMN_ERROR_PART / JSMN_ERROR_NOMEM); all three map to the same one-line error with the filename. No arbitrary parser recursion — jsmn is flat token array. |
| T-06-16 | Tampering | Unknown register name in --config JSON silently ignored | mitigate | Per D-14: EVERY top-level key must match a reflected `spu94_reg_name(i)` result. First unknown key → one-line error: `spu94: error: unknown register '%s' in '%s' — valid names: [35 names]` |
| T-06-17 | Tampering | Out-of-range integer in --config JSON hex string → UB on int16 cast | mitigate | Per D-13: use strtol(...,..., 16) for hex, strtol(...,..., 10) for integer; check vs `INT16_MIN..INT16_MAX` (signed family) or `0..0xFFFF` (unsigned family) BEFORE casting; on out-of-range: `spu94: error: value %ld for register '%s' out of range [%d..%d]`. |
| T-06-18 | Tampering | Huge --config JSON blowing past MAX_TOKENS | mitigate | MAX_TOKENS=1024 is > 10× worst case (35 reg names × 2 tokens + object markers). Overflow returns JSMN_ERROR_NOMEM, mapped to the single-line parse-error message. |
| T-06-19 | Tampering | Path traversal / symlink in --input or --output | accept | CLI runs under user's own privileges (V4 Access Control not in scope per RESEARCH). If an attacker can write to `~/foo.json` with a `--config`-valid JSON, the user already has bigger problems. No chroot applied. |
| T-06-20 | DoS | Gigantic input WAV exhausts memory | accept | CLI processes in fixed-size blocks (1024 samples × 2 channels × 2 bytes = 4 KB per block). dr_wav streams; total memory bounded. A 10-hour WAV does not consume 10 hours of RAM. |
| T-06-21 | Information Disclosure | Error messages leak filesystem paths | accept | Paths in errors are paths the user passed on argv — disclosing them back is informational, not a secret. |
| T-06-22 | Tampering | `--preset` argument matches a preset id number instead of a name | mitigate | Per D-05 error shape: only the 10 string names are accepted (case-insensitive, spaces→underscores); numeric input → `spu94: error: unknown preset 'N' — valid: off, room, ...`. Prevents accidental integer override paths. |

V5 Input Validation is the dominant category. All three input types (argv, JSON, WAV) have explicit validation paths mapping to the CLI-04 one-line-error contract.

```cmake
# src/cli/CMakeLists.txt — Phase 6 Plan 3
# Builds the spu94 CLI binary. Vendored dr_wav + jsmn are included PRIVATEly
# so they never leak to consumers that link spu94_shared (Pitfall 8).
#
# The binary is installed into the Python package directory ONLY when
# scikit-build-core is driving (SKBUILD_PROJECT_NAME defined). Plain
# `cmake -B build && cmake --build build` leaves the binary in
# build/src/cli/spu94 for dev / ctest use.

add_executable(spu94_cli
    main.c
    wav_io.c
    json_config.c
    preset_names.c
)

# Link the shared library (NOT the static) so CLI distribution matches
# wheel layout. Plan 4's install rules drop libspu94.so next to the
# binary in the wheel; $ORIGIN RPATH finds it.
target_link_libraries(spu94_cli PRIVATE spu94_shared)

# PRIVATE include dirs — dr_wav.h and jsmn.h are visible only inside
# this target's TUs; no transitive leak.
target_include_directories(spu94_cli PRIVATE
    ${CMAKE_SOURCE_DIR}/vendor/dr_wav
    ${CMAKE_SOURCE_DIR}/vendor/jsmn
)

# The CLI needs the SPU-94 public headers (same as any consumer).
target_include_directories(spu94_cli PRIVATE
    ${CMAKE_SOURCE_DIR}/include
)

# Rename to plain `spu94` — the wheel-installed binary (and the dev
# `build/src/cli/spu94` path) both need this exact name because the
# Python entry_point shim looks for a file named `spu94` (or `spu94.exe`).
set_target_properties(spu94_cli PROPERTIES OUTPUT_NAME spu94)

# Determinism + warnings — inherit from the shared warnings target.
# Vendored headers are included with -Wno-* discipline if needed;
# confirm on first build. If dr_wav or jsmn trigger -Werror warnings
# under our determinism flags, add `target_compile_options(spu94_cli
# PRIVATE -Wno-specific-flag)` TARGETED to the CLI target only —
# do NOT weaken the shared spu94_warnings interface.
target_link_libraries(spu94_cli PRIVATE spu94_warnings)

# $ORIGIN RPATH for wheel install (Plan 4 layout — libspu94.so next to
# the binary). macOS branch added for future portability.
if(APPLE)
    set(_spu94_rpath_origin "@loader_path")
else()
    set(_spu94_rpath_origin "$ORIGIN")
endif()
set_property(TARGET spu94_cli PROPERTY INSTALL_RPATH "${_spu94_rpath_origin}")

# Install rule — only when scikit-build-core is driving.

    install(TARGETS spu94_cli
            RUNTIME DESTINATION "${SKBUILD_PROJECT_NAME}"
    )
```

```c
/* One-line error printing discipline (D-05). No variadic spew. No tracebacks. */
#define SPU94_ERROR(fmt, ...) fprintf(stderr, "spu94: error: " fmt "\n", ##__VA_ARGS__)

static void print_help(void) {
    /* Tone: recording-engineer-oriented, polished. Anthony is not a coder. */
    fputs(
        "Usage: spu94 [OPTIONS] INPUT.wav OUTPUT.wav\n"
        "\n"
        "Render a WAV file through the PlayStation 1 SPU reverb.\n"
        "\n"
        "Signal choice (one of):\n"
        "  --preset <name>        Apply a factory preset. See --list-presets.\n"
        "  --config <path.json>   Apply a register-map or override from JSON.\n"
        "\n"
        "Optional:\n"
        "  --tail-seconds <N>     Append N seconds of reverb tail after input ends.\n"
        "\n"
        "Utility:\n"
        "  --list-presets         Print the 10 factory preset names and exit.\n"
        "  -h, --help             Show this message and exit.\n"
        "\n"
        "Examples:\n"
        "  spu94 --preset hall input.wav output.wav\n"
        "  spu94 --preset hall --tail-seconds 2 input.wav output.wav\n"
        "  spu94 --config my_override.json input.wav output.wav\n"
        "\n"
        "Input must be 16-bit PCM stereo at 44.1 kHz. Output is the same format.\n",
        stdout);
}

int main(int argc, char **argv) {
    static const struct option long_opts[] = {
        {"preset",       required_argument, NULL, 'p'},
        {"config",       required_argument, NULL, 'c'},
        {"tail-seconds", required_argument, NULL, 't'},
        {"list-presets", no_argument,       NULL, 'l'},
        {"help",         no_argument,       NULL, 'h'},
        {NULL, 0, NULL, 0}
    };

    const char *preset_name = NULL;
    const char *config_path = NULL;
    double tail_seconds = 0.0;
    int opt;

    while ((opt = getopt_long(argc, argv, "p:c:t:lh", long_opts, NULL)) != -1) {
        switch (opt) {
            case 'p': preset_name = optarg; break;
            case 'c': config_path = optarg; break;
            case 't': {
                char *endp = NULL;

                    SPU94_ERROR("invalid value for --tail-seconds: '%s'", optarg);
                    return 2;
                }
                break;

            case 'l':
                spu94_cli_list_presets(stdout);
                return 0;
            case 'h':
                print_help();
                return 0;
            case '?':
            default:
                SPU94_ERROR("try --help");
                return 2;
        }
    }

    if (preset_name && config_path) {
        SPU94_ERROR("--preset and --config are mutually exclusive");
        return 2;
    }
    if (!preset_name && !config_path) {
        SPU94_ERROR("one of --preset or --config is required (try --help)");
        return 2;
    }
    if (argc - optind != 2) {
        SPU94_ERROR("expected INPUT.wav OUTPUT.wav (got %d positional argument%s)",
                    argc - optind, (argc - optind) == 1 ? "" : "s");
        return 2;

    const char *in_path  = argv[optind];
    const char *out_path = argv[optind + 1];

     * WAV read + preset/config apply + spu94_process loop + tail flush +
```

**Step F — create `tests/fixtures/generate_fixtures.py`** (runs at configure time to produce `tests/fixtures/sample_1s_stereo_44k.wav`, a deterministic 1-second ramp):

```python
#!/usr/bin/env python3
"""tests/fixtures/generate_fixtures.py — Phase 6 Plan 3 Task 1.

Generates a deterministic 1-second 44.1 kHz stereo 16-bit WAV file for
CLI round-trip tests. Deterministic means: same bytes across hosts,
across Python versions, across numpy versions. Uses only stdlib + numpy.
"""
import argparse
import struct
import wave
from pathlib import Path

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", type=Path, required=True)
    args = ap.parse_args()

    sr = 44100
    n_frames = sr  # exactly 1 second
    samples = bytearray()

    # Deterministic content: linear ramps on both channels, opposite phase,
    # spanning [-8000, +8000] — avoids int16 clip, produces a non-trivial
    # input to the reverb.
    for i in range(n_frames):
        # L: ramp up; R: ramp down.
        l = int(-8000 + (16000 * i) // n_frames)
        r = int( 8000 - (16000 * i) // n_frames)
        samples.extend(struct.pack("<h", l))
        samples.extend(struct.pack("<h", r))

    args.out.parent.mkdir(parents=True, exist_ok=True)
    with wave.open(str(args.out), "wb") as w:
        w.setnchannels(2)
        w.setsampwidth(2)
        w.setframerate(sr)
        w.writeframes(bytes(samples))
    print(f"Wrote {args.out} ({n_frames} frames × 2 channels × 2 bytes = {len(samples)} bytes)")

if __name__ == "__main__":
    main()
```

```c
#include "preset_names.h"
#include <string.h>
#include <ctype.h>

/* Normalize a user-provided preset name to canonical form:
 * lowercase + spaces→underscores. Writes up to out_size-1 chars + NUL. */
static size_t normalize_name(const char *in, char *out, size_t out_size) {
    size_t i;
    for (i = 0; i + 1 < out_size && in[i]; ++i) {
        char c = in[i];
        if (c == ' ') c = '_';
        else c = (char)tolower((unsigned char)c);
        out[i] = c;
    }
    out[i] = '\0';
    return i;
}

int spu94_cli_preset_id_by_name(const char *name) {
    if (!name) return -1;
    char want[64];
    normalize_name(name, want, sizeof want);
    for (int i = 0; i < SPU94_PRESET__COUNT; ++i) {
        const char *c_name = spu94_presets[i].name;
        if (!c_name) continue;
        char canonical[64];
        normalize_name(c_name, canonical, sizeof canonical);
        if (strcmp(want, canonical) == 0) return i;
    }
    return -1;
}

const char *spu94_cli_preset_canonical_name(int preset_id) {
    /* Cache canonical names on first call — amortized O(1) lookup. */
    static char buf[SPU94_PRESET__COUNT][64];
    static int initialized = 0;
    if (!initialized) {
        for (int i = 0; i < SPU94_PRESET__COUNT; ++i) {
            const char *n = spu94_presets[i].name;
            normalize_name(n ? n : "", buf[i], sizeof buf[i]);
        }
        initialized = 1;
    }
    if (preset_id < 0 || preset_id >= SPU94_PRESET__COUNT) return NULL;
    return buf[preset_id];
}

void spu94_cli_list_presets(FILE *out) {
    for (int i = 0; i < SPU94_PRESET__COUNT; ++i) {
        const char *n = spu94_cli_preset_canonical_name(i);
        if (n) fprintf(out, "%s\n", n);
    }
}

void spu94_cli_preset_name_list(char *buf, size_t buf_size) {
    if (!buf || buf_size == 0) return;
    buf[0] = '\0';
    size_t pos = 0;
    for (int i = 0; i < SPU94_PRESET__COUNT; ++i) {
        const char *n = spu94_cli_preset_canonical_name(i);
        if (!n) continue;
        const char *sep = (i == 0) ? "" : ", ";
        int w = snprintf(buf + pos, buf_size - pos, "%s%s", sep, n);
        if (w <= 0 || (size_t)w >= buf_size - pos) return;
        pos += (size_t)w;
    }
}
```

```c
int spu94_cli_wav_load(const char *path, spu94_cli_wav_t *wav,
                      char *err_buf, size_t err_buf_size) {
    memset(wav, 0, sizeof *wav);

    /* Pre-check: file existence. Distinguishes 'not found' from 'malformed'
     * so CLI-04 error messages can be specific. */
    FILE *fp = fopen(path, "rb");
    if (!fp) {
        snprintf(err_buf, err_buf_size, "input file '%s' not found", path);
        return 1;
    }
    fclose(fp);

    drwav dw;
    if (!drwav_init_file(&dw, path, NULL)) {
        snprintf(err_buf, err_buf_size,
                 "could not read WAV file '%s' (malformed header?)", path);
        return 1;
    }

    if (dw.channels != 2) {
        snprintf(err_buf, err_buf_size,
                 "WAV file '%s' has %u channels; stereo (2 channels) required",
                 path, (unsigned)dw.channels);
        drwav_uninit(&dw);
        return 1;
    }
    if (dw.sampleRate != 44100) {
        snprintf(err_buf, err_buf_size,
                 "WAV file '%s' has sample rate %u Hz; 44100 Hz required",
                 path, (unsigned)dw.sampleRate);
        drwav_uninit(&dw);
        return 1;
    }

    wav->sample_rate = dw.sampleRate;
    wav->num_channels = dw.channels;
    wav->num_frames = dw.totalPCMFrameCount;

    /* Read as interleaved int16, then deinterleave to planar L/R. */
    size_t interleaved_bytes = (size_t)wav->num_frames * wav->num_channels * sizeof(int16_t);
    int16_t *interleaved = (int16_t *)malloc(interleaved_bytes);
    if (!interleaved) {
        drwav_uninit(&dw);
        snprintf(err_buf, err_buf_size, "out of memory loading '%s'", path);
        return 1;
    }

    drwav_uint64 got = drwav_read_pcm_frames_s16(&dw, wav->num_frames, interleaved);
    drwav_uninit(&dw);
    if (got != wav->num_frames) {
        free(interleaved);
        snprintf(err_buf, err_buf_size,
                 "failed to read all frames from '%s'", path);
        return 1;
    }

    wav->L = (int16_t *)malloc((size_t)wav->num_frames * sizeof(int16_t));
    wav->R = (int16_t *)malloc((size_t)wav->num_frames * sizeof(int16_t));
    if (!wav->L || !wav->R) {
        free(wav->L); free(wav->R); free(interleaved);
        wav->L = NULL; wav->R = NULL;
        snprintf(err_buf, err_buf_size, "out of memory deinterleaving '%s'", path);
        return 1;
    }
    for (uint64_t i = 0; i < wav->num_frames; ++i) {
        wav->L[i] = interleaved[i * 2 + 0];
        wav->R[i] = interleaved[i * 2 + 1];
    }
    free(interleaved);
    return 0;
}

int spu94_cli_wav_write(const char *path,
                       const int16_t *L, const int16_t *R,
                       uint64_t num_frames, uint32_t sample_rate,
                       char *err_buf, size_t err_buf_size) {
    drwav_data_format fmt;
    memset(&fmt, 0, sizeof fmt);
    fmt.container = drwav_container_riff;
    fmt.format = DR_WAVE_FORMAT_PCM;
    fmt.channels = 2;
    fmt.sampleRate = sample_rate;
    fmt.bitsPerSample = 16;

    drwav dw;
    if (!drwav_init_file_write(&dw, path, &fmt, NULL)) {
        snprintf(err_buf, err_buf_size,
                 "could not open '%s' for writing (permission denied?)", path);
        return 1;
    }

    /* Interleave in blocks to cap peak memory at 4096*2*2 = 16 KB. */
    const uint64_t BLOCK = 4096;
    int16_t interleaved[4096 * 2];
    for (uint64_t i = 0; i < num_frames; i += BLOCK) {
        uint64_t block = (num_frames - i < BLOCK) ? (num_frames - i) : BLOCK;
        for (uint64_t j = 0; j < block; ++j) {
            interleaved[j * 2 + 0] = L[i + j];
            interleaved[j * 2 + 1] = R[i + j];
        }
        drwav_uint64 written = drwav_write_pcm_frames(&dw, block, interleaved);
        if (written != block) {
            drwav_uninit(&dw);
            snprintf(err_buf, err_buf_size,
                     "failed to write all frames to '%s' (disk full?)", path);
            return 1;
        }
    }

    drwav_uninit(&dw);
    return 0;
}
```

```c
#define JSMN_STATIC
#include <jsmn.h>
#include "json_config.h"
#include "preset_names.h"
#include <spu94/spu94_registers.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>
#include <limits.h>

#define MAX_TOKENS 1024

/* Parse a value token as JSON integer OR hex string ("0x..." / "-0x..."). */
static int parse_int_or_hex(const char *json, const jsmntok_t *t, int32_t *out) {
    int len = t->end - t->start;
    if (len <= 0 || len >= 32) return -1;
    char buf[32];
    memcpy(buf, json + t->start, (size_t)len);
    buf[len] = '\0';
    char *endp = NULL;
    long v = 0;
    errno = 0;
    if (t->type == JSMN_STRING) {
        const char *s = buf;
        int neg = 0;
        if (*s == '-') { neg = 1; ++s; }
        if (s[0] != '0' || (s[1] != 'x' && s[1] != 'X')) return -1;
        v = strtol(s + 2, &endp, 16);
        if (neg) v = -v;
    } else if (t->type == JSMN_PRIMITIVE) {
        v = strtol(buf, &endp, 10);
    } else {
        return -1;
    }
    if (errno != 0 || !endp || *endp != '\0') return -1;
    if (v < INT_MIN || v > INT_MAX) return -1;
    *out = (int32_t)v;
    return 0;
}

static int tok_streq(const char *json, const jsmntok_t *t, const char *s) {
    int len = t->end - t->start;
    return (int)strlen(s) == len && memcmp(json + t->start, s, (size_t)len) == 0;
}

static int find_reg_by_name(const char *json, const jsmntok_t *tkey) {
    int len = tkey->end - tkey->start;
    if (len <= 0 || len >= 64) return -1;
    char buf[64];
    memcpy(buf, json + tkey->start, (size_t)len);
    buf[len] = '\0';
    for (int i = 0; i < SPU94_REG__COUNT; ++i) {
        const char *n = spu94_reg_name((spu94_reg_t)i);
        if (n && strcmp(buf, n) == 0) return i;
    }
    return -1;
}

static int apply_one(const char *json,
                    const jsmntok_t *tkey, const jsmntok_t *tval,
                    spu94_state *state,
                    char *err_buf, size_t err_buf_size) {
    int reg = find_reg_by_name(json, tkey);
    if (reg < 0) {
        int len = tkey->end - tkey->start;
        char name[64] = {0};
        memcpy(name, json + tkey->start, (size_t)(len < 63 ? len : 63));
        snprintf(err_buf, err_buf_size,
                 "unknown register '%s' (use one of the 35 canonical names like vIIR, mBASE, dCOMB1)",
                 name);
        return 1;
    }

    int32_t value;
    if (parse_int_or_hex(json, tval, &value) != 0) {
        int len = tval->end - tval->start;
        char vb[64] = {0};
        memcpy(vb, json + tval->start, (size_t)(len < 63 ? len : 63));
        snprintf(err_buf, err_buf_size,
                 "invalid value '%s' for register '%s' (expected integer or hex string)",
                 vb, spu94_reg_name((spu94_reg_t)reg));
        return 1;
    }

    spu94_reg_type_t type = spu94_reg_type((spu94_reg_t)reg);
    if (type == SPU94_REG_TYPE_I16) {
        if (value < -32768 || value > 32767) {
            snprintf(err_buf, err_buf_size,
                     "value %d for register '%s' out of range [-32768..32767]",
                     (int)value, spu94_reg_name((spu94_reg_t)reg));
            return 1;
        }
        spu94_set_reg_i16(state, (spu94_reg_t)reg, (int16_t)value);
    } else {
        if (value < 0 || value > 0xFFFF) {
            snprintf(err_buf, err_buf_size,
                     "value %d for register '%s' out of range [0..65535]",
                     (int)value, spu94_reg_name((spu94_reg_t)reg));
            return 1;
        }
        spu94_set_reg_u16(state, (spu94_reg_t)reg, (uint16_t)value);
    }
    return 0;
}

int spu94_cli_json_apply(const char *path, spu94_state *state,
                        char *err_buf, size_t err_buf_size) {
    FILE *fp = fopen(path, "rb");
    if (!fp) {
        snprintf(err_buf, err_buf_size, "config file '%s' not found", path);
        return 1;
    }
    fseek(fp, 0, SEEK_END);
    long fs = ftell(fp);
    fseek(fp, 0, SEEK_SET);
    if (fs < 0 || fs > 1024L * 1024L) {
        fclose(fp);
        snprintf(err_buf, err_buf_size,
                 "config file '%s' is empty or larger than 1 MB", path);
        return 1;
    }
    char *json = (char *)malloc((size_t)fs + 1);
    if (!json) {
        fclose(fp);
        snprintf(err_buf, err_buf_size, "out of memory reading '%s'", path);
        return 1;
    }
    size_t got = fread(json, 1, (size_t)fs, fp);
    fclose(fp);
    json[got] = '\0';

    jsmn_parser p;
    jsmntok_t tokens[MAX_TOKENS];
    jsmn_init(&p);
    int n = jsmn_parse(&p, json, got, tokens, MAX_TOKENS);
    if (n < 0 || n == 0 || tokens[0].type != JSMN_OBJECT) {
        free(json);
        snprintf(err_buf, err_buf_size,
                 "invalid JSON in '%s' (expected top-level object)", path);
        return 1;
    }

    /* Auto-detect shape: presence of "base" key at top level. */
    int has_base = 0;
    {
        int cursor = 1;
        int pairs = tokens[0].size;
        for (int k = 0; k < pairs && cursor + 1 < n; ++k) {
            if (tokens[cursor].type == JSMN_STRING &&
                tok_streq(json, &tokens[cursor], "base")) {
                has_base = 1;
                break;
            }
            cursor += 2;
            /* If the value was an object/array, skip its children. */
            /* jsmn lays out children contiguously; tokens[value].size tells
             * us how many immediate children to skip. This handles nested
             * "overrides" objects. */
            if (cursor - 1 < n && tokens[cursor - 1].size > 0) {
                /* simplistic walk — works because our schema is flat or
                 * one-level nested only. */
                int child_pairs = tokens[cursor - 1].size;
                cursor += 2 * child_pairs;
            }
        }
    }

    if (has_base) {
        /* Override shape. */
        const jsmntok_t *base_val = NULL;
        const jsmntok_t *overrides = NULL;
        int cursor = 1;
        int pairs = tokens[0].size;
        for (int k = 0; k < pairs && cursor + 1 < n; ++k) {
            const jsmntok_t *tkey = &tokens[cursor];
            const jsmntok_t *tval = &tokens[cursor + 1];
            if (tok_streq(json, tkey, "base")) base_val = tval;
            else if (tok_streq(json, tkey, "overrides")) overrides = tval;
            cursor += 2;
            if (tval->type == JSMN_OBJECT || tval->type == JSMN_ARRAY) {
                cursor += 2 * tval->size;
            }
        }
        if (!base_val || base_val->type != JSMN_STRING) {
            free(json);
            snprintf(err_buf, err_buf_size,
                     "'%s' has \"base\" but value is not a preset name string", path);
            return 1;
        }
        int name_len = base_val->end - base_val->start;
        char bname[64] = {0};
        memcpy(bname, json + base_val->start, (size_t)(name_len < 63 ? name_len : 63));
        int pid = spu94_cli_preset_id_by_name(bname);
        if (pid < 0) {
            char names[256]; spu94_cli_preset_name_list(names, sizeof names);
            free(json);
            snprintf(err_buf, err_buf_size,
                     "unknown preset '%s' in '%s' — valid: %s", bname, path, names);
            return 1;
        }
        spu94_result_t rc = spu94_load_preset(state, (spu94_preset_id_t)pid);
        if (rc != SPU94_OK) {
            free(json);
            snprintf(err_buf, err_buf_size,
                     "load_preset failed for base '%s' (code %d)", bname, (int)rc);
            return 1;
        }
        if (overrides) {
            if (overrides->type != JSMN_OBJECT) {
                free(json);
                snprintf(err_buf, err_buf_size,
                         "'%s' \"overrides\" must be an object", path);
                return 1;
            }
            int ov_idx = (int)(overrides - tokens);
            int ov_pairs = overrides->size;
            int cur = ov_idx + 1;
            for (int k = 0; k < ov_pairs && cur + 1 < n; ++k) {
                const jsmntok_t *tkey = &tokens[cur];
                const jsmntok_t *tval = &tokens[cur + 1];
                if (apply_one(json, tkey, tval, state, err_buf, err_buf_size)) {
                    free(json);
                    return 1;
                }
                cur += 2;
            }
        }
    } else {
        /* Flat shape — all 35 registers required. */
        int pairs = tokens[0].size;
        if (pairs != SPU94_REG__COUNT) {
            free(json);
            snprintf(err_buf, err_buf_size,
                     "flat config '%s' must specify all %d registers (found %d)",
                     path, SPU94_REG__COUNT, pairs);
            return 1;
        }
        int cur = 1;
        for (int k = 0; k < pairs && cur + 1 < n; ++k) {
            const jsmntok_t *tkey = &tokens[cur];
            const jsmntok_t *tval = &tokens[cur + 1];
            if (apply_one(json, tkey, tval, state, err_buf, err_buf_size)) {
                free(json);
                return 1;
            }
            cur += 2;
        }
    }

    free(json);
    return 0;
}
```
