```c
#include "spu94_state_internal.h"

/* 64-bit bitmask: bit i set = register i is unsigned (u16); bit i clear = signed (i16).
 * 35 bits used; upper 29 reserved. Hand-maintained per the signedness table in the
 * interfaces block of this plan. */
static const uint64_t spu94_reg_is_u16_mask =
    (UINT64_C(1) << SPU94_REG_mBASE) |
    (UINT64_C(1) << SPU94_REG_dAPF1) |
    (UINT64_C(1) << SPU94_REG_dAPF2) |
    (UINT64_C(1) << SPU94_REG_mLSAME) |
    (UINT64_C(1) << SPU94_REG_mRSAME) |
    (UINT64_C(1) << SPU94_REG_mLCOMB1) |
    (UINT64_C(1) << SPU94_REG_mRCOMB1) |
    (UINT64_C(1) << SPU94_REG_mLCOMB2) |
    (UINT64_C(1) << SPU94_REG_mRCOMB2) |
    (UINT64_C(1) << SPU94_REG_dLSAME) |
    (UINT64_C(1) << SPU94_REG_dRSAME) |
    (UINT64_C(1) << SPU94_REG_mLDIFF) |
    (UINT64_C(1) << SPU94_REG_mRDIFF) |
    (UINT64_C(1) << SPU94_REG_mLCOMB3) |
    (UINT64_C(1) << SPU94_REG_mRCOMB3) |
    (UINT64_C(1) << SPU94_REG_mLCOMB4) |
    (UINT64_C(1) << SPU94_REG_mRCOMB4) |
    (UINT64_C(1) << SPU94_REG_dLDIFF) |
    (UINT64_C(1) << SPU94_REG_dRDIFF) |
    (UINT64_C(1) << SPU94_REG_mLAPF1) |
    (UINT64_C(1) << SPU94_REG_mRAPF1) |
    (UINT64_C(1) << SPU94_REG_mLAPF2) |
    (UINT64_C(1) << SPU94_REG_mRAPF2);
/* Verify the mask has exactly 23 bits set (hand-audit convenience). */

spu94_reg_type_t spu94_reg_type(spu94_reg_t reg) {
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return SPU94_REG_TYPE_I16;  /* arbitrary */
    return (spu94_reg_is_u16_mask & (UINT64_C(1) << reg)) ? SPU94_REG_TYPE_U16 : SPU94_REG_TYPE_I16;
}

/* Policy lookup — defined in spu94_write_policy.c, Task 2. Forward-declared here. */
typedef enum { SPU94_WRITE_IMMEDIATE = 0, SPU94_WRITE_TICK_LATCHED = 1 } spu94_write_policy_t;
extern const spu94_write_policy_t spu94_write_policy_table[SPU94_REG__COUNT];

/* mBASE side-effect handler — Plan 04 replaces this stub with the snap-on-write
 * behavior (ADR-0006). Plan 03 calls it; Plan 04 implements it. */
void spu94_mbase_on_write(spu94_state *state, uint16_t new_mbase);  /* fwd-decl */

spu94_result_t spu94_set_reg_i16(spu94_state *state, spu94_reg_t reg, int16_t value) {
    if (!state) return SPU94_UNKNOWN_REG;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return SPU94_UNKNOWN_REG;
    if (spu94_reg_type(reg) != SPU94_REG_TYPE_I16) return SPU94_TYPE_MISMATCH;

    /* All 12 i16 registers are v*-prefix and policy is IMMEDIATE. */
    state->reg_values[reg] = value;
    state->pending_values[reg] = value;  /* keep pending in sync for IMMEDIATE (D-06) */
    state->pending_mask &= ~(UINT64_C(1) << reg);
    return SPU94_OK;
}

spu94_result_t spu94_set_reg_u16(spu94_state *state, spu94_reg_t reg, uint16_t value) {
    if (!state) return SPU94_UNKNOWN_REG;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return SPU94_UNKNOWN_REG;
    if (spu94_reg_type(reg) != SPU94_REG_TYPE_U16) return SPU94_TYPE_MISMATCH;

    int16_t stored = (int16_t)value;  /* bitwise reinterpret — the reg_values array is int16 storage */

    spu94_write_policy_t policy = spu94_write_policy_table[reg];
    if (policy == SPU94_WRITE_IMMEDIATE) {
        /* mBASE is the only IMMEDIATE u16 register. Other u16s are TICK_LATCHED. */
        state->reg_values[reg] = stored;
        state->pending_values[reg] = stored;
        state->pending_mask &= ~(UINT64_C(1) << reg);
        if (reg == SPU94_REG_mBASE) {
            spu94_mbase_on_write(state, value);  /* Plan 04's stub; fires the snap */
        }
    } else {
        /* TICK_LATCHED: pending_values captures, pending_mask sets; active unchanged. */
        state->pending_values[reg] = stored;
        state->pending_mask |= (UINT64_C(1) << reg);
    }
    return SPU94_OK;
}

int16_t spu94_get_reg_i16(const spu94_state *state, spu94_reg_t reg) {
    if (!state) return 0;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return 0;
    return state->reg_values[reg];
}

uint16_t spu94_get_reg_u16(const spu94_state *state, spu94_reg_t reg) {
    if (!state) return 0;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return 0;
    return (uint16_t)state->reg_values[reg];
}

int16_t spu94_get_reg_i16_pending(const spu94_state *state, spu94_reg_t reg) {
    if (!state) return 0;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return 0;
    return state->pending_values[reg];
}

uint16_t spu94_get_reg_u16_pending(const spu94_state *state, spu94_reg_t reg) {
    if (!state) return 0;
    if ((int)reg < 0 || (int)reg >= (int)SPU94_REG__COUNT) return 0;
    return (uint16_t)state->pending_values[reg];
}
```

```c
#include <stdint.h>

/* Apply every bit set in state->pending_mask. Copy pending_values[i] to
 * reg_values[i] and clear the bit. Called exactly once from spu94_tick
 * at tick start (Pitfall 4 — exactly one call site). */
void spu94_apply_pending_writes(spu94_state *state) {
    if (!state) return;
    uint64_t mask = state->pending_mask;
    /* Iterate set bits. Simple loop is fine — at most 35 bits. */
    for (int i = 0; i < (int)SPU94_REG__COUNT; ++i) {
        if (mask & (UINT64_C(1) << i)) {
            state->reg_values[i] = state->pending_values[i];
        }
    }
    state->pending_mask = 0;
}
```

```markdown
## ADR-0005: Per-register mid-stream write-timing policy — split policy with swappable table

**Status:** Accepted (2026-04-19, Phase 2)

**Context:**

nocash's psx-spx documentation is silent on what happens when an SPU reverb
register is written mid-tick — i.e., during the 22.05 kHz stereo tick in which
the reverb network is computing. The Sony BIOS procedure for setting up a new
reverb effect is to disable the reverb, write every register, and then re-
enable, which implies mid-stream writes are not a spec'd operation. SPU-94
nonetheless has to pick a defensible behavior because (a) real-time modulation
is a first-class use case per PROJECT.md, and (b) the algorithm must not crash
or corrupt memory on a mid-stream write.

The 35 registers fall into two structural families:

1. Gain-type registers (`v*`-prefix: vLOUT, vROUT, vIIR, vCOMB1..4, vWALL,
   vAPF1, vAPF2, vLIN, vRIN — 12 total) participate in per-sample multiplies.
   A faithful model of the hardware multiplier reads the register at the
   multiply site; a mid-tick write is visible to the next multiply.

2. Address/delay-type registers (`d*` / `m*`-prefix — 22 total) index into
   the reverb work buffer. The reverb algorithm's correctness depends on a
   consistent pair of addresses across the L and R half-cycles of a stereo
   tick. A mid-tick change would corrupt the L/R relationship.

`mBASE` is a third case: it is `u16` but its timing policy is IMMEDIATE (the
config value itself is instant) with a stateful side effect (the BufferAddress
snap, resolved separately in ADR-0006).

**Decision:**

SPU-94 ships a split policy:

- **IMMEDIATE** for all 12 `v*` gain-type registers AND for `mBASE`.
  Writes become visible to the next register read and (for `v*`) the next
  multiply, within the same tick.
- **TICK_LATCHED** for the remaining 22 `d*` / `m*` address/delay registers.
  Writes stage into a shadow slot (`state->pending_values[i]`) and are
  applied atomically at the start of the next `spu94_tick()` call, before
  any buffer-address advance or reverb computation.

The policy is implemented as a `static const spu94_write_policy_t`
35-entry array keyed by `spu94_reg_t`, located in
`src/spu94/spu94_write_policy.c`. This array IS the swappable seam (D-05):
the future SPU-94 Controllers milestone pins an alternative table to
explore "what if all writes are IMMEDIATE" or similar without modifying
core engine code.

The pending-value shadow is observable to callers via
`spu94_get_reg_{i16,u16}_pending`, which returns what will be applied at
the next tick. For IMMEDIATE-policy registers the pending and active
readings always match.

**Consequences:**

- **Assumption flagged:** The exact per-register assignment (all `v*`
  IMMEDIATE, all `d*`/`m*` TICK_LATCHED) is not spec-backed. It is
  defensible on structural grounds but could be adjusted if hardware-
  witness evidence (Phase 7) contradicts. The seam exists precisely so
  that adjustment is cheap.

- **Test obligation:** Plan 05 adds per-register policy tests
  (`tests/unit/registers/test_register_policy.c`) that exercise:
  - Write → `get == get_pending == value` for every `v*` register.
  - Write → `get != get_pending` immediately for every `d*`/`m*` register,
    followed by `tick()` → `get == get_pending == value` after.
  - mBASE specifically: IMMEDIATE update of the config value (side effect
    is in ADR-0006's scope, tested in `test_buffer_mbase.c`).

- **Pitfall 4 protection:** `spu94_apply_pending_writes()` is called
  from exactly one location — the first line of `spu94_tick()`. Any
  future change that calls it from another site violates the contract.
  A grep-based CI check could be added if drift becomes a concern.

- **Revision paths:**
  - Hardware witness results (Milestone 5) may show that some `v*`
    registers are actually TICK_LATCHED on the real chip; the policy
    table changes with a new ADR superseding the assignment.
  - Controllers may want to add a third policy (CROSSFADE, for zipper-
    free gain modulation) — additive ADR; new enum value; new rows in
    the table; engine setter switches on three cases instead of two.

**Sources:**

  enable" convention).
- Prior ADR: ADR-0004 (extensibility taps) — same swappable-seam
  principle applied to a different piece of the API.
```
