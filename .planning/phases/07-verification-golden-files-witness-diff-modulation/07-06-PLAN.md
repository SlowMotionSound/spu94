    Step 2 — Write `scripts/check_bibliography_refs.py`:

    ```python
    #!/usr/bin/env python3
    """DOCS-03: Cross-reference validator for BIB-nnn citations.

    Parses docs/DECISIONS.md for every `BIB-nnn` reference; parses
    docs/BIBLIOGRAPHY.md for every `### BIB-nnn:` entry header; exits
    non-zero if any cited BIB-nnn has no matching entry.

    Threat T-07-06-A (citation scope creep): reject references of the form
    BIB-<non-3-digit> to keep the ID space narrow and auditable.
    """
    import argparse
    import re
    import sys
    from pathlib import Path

    BIB_REF = re.compile(r'\bBIB-(\d{3})\b')
    BIB_ENTRY = re.compile(r'^### BIB-(\d{3})(?:\s*:|\s*[—-])', re.MULTILINE)

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("--decisions", default="docs/DECISIONS.md")
        ap.add_argument("--bibliography", default="docs/BIBLIOGRAPHY.md")
        args = ap.parse_args()

        dec_text = Path(args.decisions).read_text()
        bib_text = Path(args.bibliography).read_text()

        cited = set(BIB_REF.findall(dec_text))
        defined = set(BIB_ENTRY.findall(bib_text))

        dangling = sorted(cited - defined)
        unused = sorted(defined - cited)  # informational only

        if dangling:
            print(f"FAIL: {len(dangling)} BIB-nnn reference(s) not defined in "
                  f"{args.bibliography}:", file=sys.stderr)
            for d in dangling:
                print(f"  BIB-{d}", file=sys.stderr)
            return 1

        print(f"PASS: {len(cited)} BIB-nnn reference(s) all resolve; "
              f"{len(unused)} defined but unused (informational)")
        if unused:
            for u in unused[:5]:
                print(f"  unused: BIB-{u}")
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```
