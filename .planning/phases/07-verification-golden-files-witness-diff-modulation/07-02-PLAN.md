    #!/usr/bin/env python3
    """scripts/regenerate_goldens.py — TEST-04 + TEST-08 deterministic golden generator.

    D-09 .wav + .sha256 sidecar pair; D-10 tests/golden/<preset>/<input>.{wav,sha256};
    D-11 five inputs (impulse/white_noise/sine_1khz/silence/sweep); D-12 regeneration
    requires ADR.

    Modes:
      default (no args) — regenerate all 50 goldens + sidecars, overwriting.
      --check          — for each existing golden, re-render and compare SHA-256
                         to committed sidecar; exit 1 on any mismatch.
    """
    import argparse, hashlib, os, subprocess, sys, tempfile
    from pathlib import Path
    import numpy as np
    from scipy.io import wavfile
    from scipy.signal import chirp

    # Determinism env
    os.environ.setdefault("LC_ALL", "C")
    os.environ.setdefault("TZ", "UTC")
    os.environ.setdefault("SOURCE_DATE_EPOCH", "1704067200")  # 2024-01-01 UTC

    # Closed allowlist — these come from spu94_preset_id_t; no external input.
    PRESETS = ["off", "room", "studio_a", "studio_b", "studio_c",
               "hall", "half_echo", "space_echo", "echo", "delay"]
    INPUTS = ["impulse", "white_noise", "sine_1khz", "silence", "sweep"]

    FS = 44100
    DURATION_SEC = 2.0
    N = int(FS * DURATION_SEC)  # 88200
    AMP = 16000
    SEED = 0x1094_DADA

    GOLDEN_ROOT = Path("tests/golden")

    def generate_input(name: str) -> np.ndarray:
        """Returns (N, 2) int16 array per the D-11 standard input set."""
        if name == "impulse":
            x = np.zeros((N, 2), dtype=np.int16)
            x[0, 0] = x[0, 1] = AMP
            return x
        if name == "silence":
            return np.zeros((N, 2), dtype=np.int16)
        if name == "white_noise":
            rng = np.random.default_rng(SEED)
            return rng.integers(-AMP, AMP, size=(N, 2), dtype=np.int16)
        if name == "sine_1khz":
            t = np.arange(N) / FS
            y = (AMP * np.sin(2 * np.pi * 1000.0 * t)).astype(np.int16)
            return np.stack([y, y], axis=1)
        if name == "sweep":
            t = np.arange(N) / FS
            y = (AMP * chirp(t, 20.0, DURATION_SEC, 20000.0,
                             method="logarithmic")).astype(np.int16)
            return np.stack([y, y], axis=1)
        raise ValueError(f"unknown input: {name}")

    def render_golden(preset: str, input_name: str, out_path: Path,
                      tmp_in: Path) -> None:
        x = generate_input(input_name)
        wavfile.write(str(tmp_in), FS, x)
        # Locate spu94 CLI binary: prefer build tree, fall back to PATH
        spu94_bin = os.environ.get("SPU94_BIN",
                                    "build/src/cli/spu94"
                                    if Path("build/src/cli/spu94").exists()
                                    else "spu94")
        r = subprocess.run(
            [spu94_bin, "--preset", preset, str(tmp_in), str(out_path)],
            check=False, capture_output=True, env={**os.environ,
                                                    "LC_ALL": "C", "TZ": "UTC"}
        )
        if r.returncode != 0:
            raise RuntimeError(f"{spu94_bin} failed on {preset}/{input_name}: "
                               f"{r.stderr.decode()[:200]}")

    def sha256_of(path: Path) -> str:
        h = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(65536), b""):
                h.update(chunk)
        return h.hexdigest()

    def main() -> int:
        ap = argparse.ArgumentParser()
        ap.add_argument("--check", action="store_true",
                        help="verify existing goldens' sidecars match fresh renders")
        args = ap.parse_args()

        failures = []
        with tempfile.TemporaryDirectory() as tmpd:
            tmp_in = Path(tmpd) / "input.wav"
            for preset in PRESETS:
                preset_dir = GOLDEN_ROOT / preset
                preset_dir.mkdir(parents=True, exist_ok=True)
                for input_name in INPUTS:
                    out_wav = preset_dir / f"{input_name}.wav"
                    out_sha = preset_dir / f"{input_name}.wav.sha256"
                    if args.check:
                        # Render to a scratch file, hash, compare
                        scratch = Path(tmpd) / f"{preset}_{input_name}.wav"
                        render_golden(preset, input_name, scratch, tmp_in)
                        fresh = sha256_of(scratch)
                        committed = out_sha.read_text().strip().split()[0] \
                                      if out_sha.exists() else ""
                        if fresh != committed:
                            failures.append(f"{preset}/{input_name}: "
                                             f"committed={committed[:16]}... "
                                             f"fresh={fresh[:16]}...")
                    else:
                        render_golden(preset, input_name, out_wav, tmp_in)
                        digest = sha256_of(out_wav)
                        out_sha.write_text(f"{digest}  {input_name}.wav\n")
        if failures:
            for f in failures:
                print(f"FAIL: {f}", file=sys.stderr)
            return 1
        if args.check:
            print(f"PASS: {len(PRESETS) * len(INPUTS)}/{len(PRESETS) * len(INPUTS)} "
                  f"goldens match")
        else:
            print(f"Regenerated {len(PRESETS) * len(INPUTS)} goldens under "
                  f"{GOLDEN_ROOT}")
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```

    """Structural meta-test: tests/golden/ has full 50-golden matrix."""
    import re
    from pathlib import Path
    import hashlib
    import pytest

    PRESETS = ["off", "room", "studio_a", "studio_b", "studio_c",
               "hall", "half_echo", "space_echo", "echo", "delay"]
    INPUTS = ["impulse", "white_noise", "sine_1khz", "silence", "sweep"]
    ROOT = Path("tests/golden")

    @pytest.mark.parametrize("preset", PRESETS)
    @pytest.mark.parametrize("input_name", INPUTS)
    def test_wav_exists(preset, input_name):
        assert (ROOT / preset / f"{input_name}.wav").is_file()

    @pytest.mark.parametrize("preset", PRESETS)
    @pytest.mark.parametrize("input_name", INPUTS)
    def test_sidecar_exists_and_format(preset, input_name):
        p = ROOT / preset / f"{input_name}.wav.sha256"
        assert p.is_file()
        text = p.read_text()
        assert re.match(r'^[0-9a-f]{64}\s+\S+\s*$', text), \
            f"sidecar {p} malformed: {text!r}"

    @pytest.mark.parametrize("preset,input_name", [
        ("hall", "impulse"), ("room", "sine_1khz"), ("studio_a", "sweep")])
    def test_sidecar_matches_wav(preset, input_name):
        wav = ROOT / preset / f"{input_name}.wav"
        sha_file = ROOT / preset / f"{input_name}.wav.sha256"
        digest = hashlib.sha256(wav.read_bytes()).hexdigest()
        sidecar_digest = sha_file.read_text().split()[0]
        assert digest == sidecar_digest, \
            f"{preset}/{input_name}: wav={digest[:16]} sidecar={sidecar_digest[:16]}"
    ```
