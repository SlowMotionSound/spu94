    #!/usr/bin/env bash
    # scripts/ci/witness_diff_build.sh — build lv2-psx-reverb fresh at pinned SHA (D-05).
    #
    # D-08 discipline: lv2 is a behavioral witness binary only; source is NOT
    # read as primary material. This script compiles the bundle and confirms the
    # exposed URIs/ports. The witness-diff Python harness reads ONLY the bundle's
    # produced audio output, never the .c/.h sources.
    #
    # Pin provenance: RESEARCH.md verified via GitHub commits view 2026-04-23;
    # master tip, no release tags. Re-verify at execute time and update if the
    # master tip has advanced (ADR required per D-05/D-08 discipline).

    set -euo pipefail

    LV2_COMMIT="${LV2_COMMIT:-424e1e8ee7f780106b005011b036386513c61db3}"
    LV2_REPO="${LV2_REPO:-https://github.com/ipatix/lv2-psx-reverb}"
    WORK="${WITNESS_WORK:-$(pwd)/.artifacts/lv2-psx-reverb}"

    rm -rf "$WORK"
    mkdir -p "$(dirname "$WORK")"

    echo "Cloning $LV2_REPO at $LV2_COMMIT..."
    git clone --quiet "$LV2_REPO" "$WORK/src"
    git -C "$WORK/src" checkout --quiet "$LV2_COMMIT"

    # Supply-chain gate: verify the commit we're on matches the pin
    ACTUAL=$(git -C "$WORK/src" rev-parse HEAD)
    if [ "$ACTUAL" != "$LV2_COMMIT" ]; then
        echo "FAIL: expected commit $LV2_COMMIT, got $ACTUAL" >&2
        exit 1
    fi

    echo "Building LV2 bundle..."
    make -C "$WORK/src" 2>&1 | tail -20

    # Install bundle to WORK/lv2/ so LV2_PATH can point at a clean tree
    mkdir -p "$WORK/lv2"
    # lv2-psx-reverb's Makefile produces a .lv2 directory; locate it
    BUNDLE=$(find "$WORK/src" -maxdepth 2 -name "*.lv2" -type d | head -1)
    if [ -z "$BUNDLE" ]; then
        echo "FAIL: no .lv2 bundle dir found in $WORK/src" >&2
        exit 1
    fi
    cp -r "$BUNDLE" "$WORK/lv2/"

    echo "LV2_PATH=$WORK/lv2"
    echo ""
    echo "=== lv2info discovery (D-08: binary output only, not source read) ==="
    LV2_PATH="$WORK/lv2" lv2info \
        "$(find "$WORK/lv2" -name "manifest.ttl" -exec grep -l "a lv2:Plugin" {} \; | head -1 | xargs dirname)" \
        2>&1 | head -80 || true

    # Emit the discovered LV2_PATH for the caller (witness_diff.py reads it)
    echo "$WORK/lv2" > "$WORK/.LV2_PATH"
    echo "PASS: lv2-psx-reverb built at $LV2_COMMIT; LV2_PATH emitted to $WORK/.LV2_PATH"
    ```

       The harness code has a small dispatcher that picks the case based on `lv2info` output parsing (which is a structured output format lv2 devs designed for exactly this purpose). A helper `def invoke_lv2apply(preset_id, in_wav, out_wav)` centralizes the branching.

       a. Reuse `generate_input()` from `scripts/regenerate_goldens.py` (or copy the 5-function body inline with the same seed/params to avoid cross-script import fragility).

       c. Render SPU-94 output: shell out to `build/src/cli/spu94 --preset <name> <in.wav> <out_spu94.wav>`.
       d. Render lv2 output: invoke_lv2apply(preset_id, <in.wav>, <out_lv2.wav>) with `LV2_PATH` set.
       e. Load both outputs via `scipy.io.wavfile.read`; normalize to float64 in [−1, 1) (int16 → `/ 32768.0`, float32 → cast; stereo → mono via `mean(axis=1)`).
       f. Cross-correlate on the broadband signal via `scipy.signal.correlate(spu, lv2, mode='full', method='fft')`; `lag = int(np.argmax(xc) - (len(lv2) - 1))`; trim both to aligned window.
       g. Build SOS filters once at module scope: `_sos_lp = butter(8, 10000/22050, btype='low', output='sos')`; `_sos_hp = butter(8, 10000/22050, btype='high', output='sos')`. Apply via `sosfiltfilt(..., padlen=1024)` to each aligned signal.
       h. Compute `lp_diff_rms = sqrt(mean((spu_lp - lv2_lp)**2))`; `lp_ref_rms = sqrt(mean(lv2_lp**2)) + 1e-12`; `lp_dbfs = 20 * log10(lp_diff_rms / lp_ref_rms)`. Same for high band.

Writes `.artifacts/witness_report.json` with schema `[{"preset": ..., "input": ..., "alignment_lag_samples": int, "low_band_diff_dbfs": float, "high_band_diff_dbfs": float, "lv2_commit": "424e1e8...", "n_samples_compared": int}, ...]`. Entries sorted by (preset, input) for deterministic output. Float values rounded to 6 decimals to avoid platform-specific IEEE differences in the last bits leaking into the "determinism" property (belt-and-suspenders — sosfiltfilt + FFT correlate are already deterministic on x86-64, but explicit rounding is cheap insurance).

    """D-06 measurement determinism gate: the metric itself must be reproducible.

    Running the harness twice on the same source must produce byte-identical
    witness_report.json (modulo nothing — the harness writes no timestamps).
    """
    import json
    import shutil
    import subprocess
    import sys
    from pathlib import Path
    import pytest

    REPORT = Path(".artifacts/witness_report.json")


    @pytest.fixture(scope="module")
    def fresh_witness_run(tmp_path_factory):
        """Run witness_diff.py once; cache the first run's report."""
        # Precondition: lv2 bundle already built (witness_diff_build.sh run
        # by the calling CI job or by the user before running pytest).
        assert Path(".artifacts/lv2-psx-reverb/.LV2_PATH").exists(), \
            "lv2 bundle not built — run scripts/ci/witness_diff_build.sh first"

        # First run
        r1 = subprocess.run(["python3", "scripts/ci/witness_diff.py"],
                            capture_output=True)
        assert r1.returncode == 0, r1.stderr.decode()[-500:]
        assert REPORT.exists()
        first = REPORT.read_bytes()

        # Second run (same everything)
        r2 = subprocess.run(["python3", "scripts/ci/witness_diff.py"],
                            capture_output=True)
        assert r2.returncode == 0, r2.stderr.decode()[-500:]
        second = REPORT.read_bytes()

        return first, second


    def test_witness_diff_is_deterministic(fresh_witness_run):
        first, second = fresh_witness_run
        if first == second:
            return  # byte-identical — best case
        # Fall back: numerical near-equality (platform FP last-bit variance tolerance)
        f = json.loads(first)
        s = json.loads(second)
        assert len(f) == len(s)
        for a, b in zip(f, s):
            assert a["preset"] == b["preset"] and a["input"] == b["input"]
            assert abs(a["low_band_diff_dbfs"] - b["low_band_diff_dbfs"]) < 1e-9
            assert abs(a["high_band_diff_dbfs"] - b["high_band_diff_dbfs"]) < 1e-9
            assert a["alignment_lag_samples"] == b["alignment_lag_samples"]


    def test_witness_report_has_50_entries(fresh_witness_run):
        first, _ = fresh_witness_run
        r = json.loads(first)
        assert len(r) == 50
        required = {"preset", "input", "alignment_lag_samples",
                    "low_band_diff_dbfs", "high_band_diff_dbfs"}
        for entry in r:
            assert required <= set(entry.keys()), \
                f"entry missing keys: {entry}"
    ```
