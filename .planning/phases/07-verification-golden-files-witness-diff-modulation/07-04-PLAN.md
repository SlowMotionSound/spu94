  <rationale>
    D-17 coverage note (rate axis, deliberate 105-case count):

    D-17 calls for stability + determinism "at every modulation rate from
    slow (~0.1 Hz) through audio-rate (~11 kHz)". A naive reading would
    parametrize sine mode across all 9 entries of RATES_HZ = [0.1, 1, 10,
    100, 500, 1000, 2000, 5000, 11025], giving 35 × 9 + 35 × 2 = 385 cases.

    Instead, the harness uses sine mode at one representative mid-rate
    (RATES_HZ[4] = 500 Hz) per (reg, mode) pair, giving 35 × 3 = 105 cases.
    This is deliberate, not a coverage reduction:

    - Sweep mode walks 0.1 Hz → 11 kHz CONTINUOUSLY over 1 second via
      scipy.signal.chirp(..., method='logarithmic'). Every register is
      driven through the full D-17 rate range inside the sweep-mode case.
      The stability axis ("no NaN, no overflow, no crash, bounded output
      at every rate") is therefore exercised per-register across the
      entire 0.1 Hz → 11 kHz surface by sweep mode alone.
    - Determinism (bit-exact repeatability) is a per-seed property, not a
      per-rate property — one seed proves it holds; running at 9 rates
      would prove it 9 times for no new information.
    - Sine mode at one mid-rate gives an additional spot-check that the
      stability invariants hold under steady-state (not-swept) modulation,
      catching any case where the sweep's continuous motion masks a
      bug that only manifests at a fixed rate.
    - Random-walk mode adds non-periodic stressing of the write-policy
      machinery (Phase 2 D-08 verification).

    Total: 35 regs × 3 modes × 2 gates (stability + determinism) + 1
    report-writer = 211 pytest cases, comfortably within the 180s
    VALIDATION.md feedback-latency budget. Extending sine mode to 3 rates
    (175 stability cases + 175 determinism cases = 350, still <180s) is
    cheap insurance but adds no new D-17 coverage beyond what sweep mode
    already walks — hence the deliberate 105-case shape.
  </rationale>

    ```python
    """Phase 7 TEST-05: modulation harness helpers.

    D-17 gate: stability + determinism at every modulation rate (0.1 Hz → 11 kHz).
    D-18 SC-4 reinterpretation: zipper at audio-rate is character, not failure.
    D-19: all 35 registers × {sine, sweep, random_walk}.
    """
    from dataclasses import dataclass
    from pathlib import Path
    import json
    import numpy as np
    import spu94
    from spu94 import SPU94, Register

    FS = 44100
    DURATION_SEC = 1.0
    N = int(FS * DURATION_SEC)
    AMP = 16000
    SEED = 0x1094_DADA
    BLOCK = 1024
    RATES_HZ = [0.1, 1.0, 10.0, 100.0, 500.0, 1000.0, 2000.0, 5000.0, 11025.0]
    SWEEP_RANGE = (0.1, 11025.0)

    # Signedness allowlist from spu94_register_io.c's 64-bit packed mask.
    # 12 v-prefix registers are signed i16; the rest are u16.
    I16_REG_NAMES = frozenset([
        "vLOUT", "vROUT", "vIIR", "vWALL",
        "vCOMB1", "vCOMB2", "vCOMB3", "vCOMB4",
        "vAPF1", "vAPF2", "vLIN", "vRIN",
    ])

    # D-08 write policy (Phase 2 Plan 03): 13 IMMEDIATE + 22 TICK_LATCHED.
    IMMEDIATE_REG_NAMES = frozenset([
        "vLOUT", "vROUT", "mBASE", "vIIR", "vWALL",
        "vCOMB1", "vCOMB2", "vCOMB3", "vCOMB4",
        "vAPF1", "vAPF2", "vLIN", "vRIN",
    ])

    def reg_is_signed(reg: Register) -> bool:
        return reg.name in I16_REG_NAMES

    def reg_valid_range(reg: Register) -> tuple[int, int]:
        if reg_is_signed(reg):
            return (-32768, 32767)
        return (0, 65535)

    def modulation_stream(mode: str, rate_hz: float, n: int, seed: int) -> np.ndarray:
        rng = np.random.default_rng(seed)
        t = np.arange(n) / FS
        if mode == "sine":
            # Normalize to ±1 then scale per register
            return (0.95 * np.sin(2 * np.pi * rate_hz * t))
        elif mode == "sweep":
            from scipy.signal import chirp
            return 0.95 * chirp(t, SWEEP_RANGE[0], DURATION_SEC, SWEEP_RANGE[1],
                                method="logarithmic")
        elif mode == "random_walk":
            steps = rng.integers(-500, 500, size=n)
            walk = np.cumsum(steps).astype(np.float64)
            m = np.max(np.abs(walk)) + 1.0
            return 0.95 * walk / m
        raise ValueError(f"unknown mode: {mode}")

    def modulation_to_register_values(norm: np.ndarray, reg: Register) -> np.ndarray:
        lo, hi = reg_valid_range(reg)
        center = (lo + hi) // 2
        half = (hi - lo) // 2
        vals = (center + norm * half).astype(np.int64)
        return np.clip(vals, lo, hi)

    def set_reg_typed(state, reg: Register, value: int) -> None:
        v = int(value)
        if reg_is_signed(reg):
            spu94.set_reg_i16(state, reg, v)
        else:
            spu94.set_reg_u16(state, reg, v)

    @dataclass
    class ModulationResult:
        reg_name: str
        mode: str
        rate_hz: float | None
        out_sha256: str  # for determinism-gate bit-identity check
        zipper_onset_hz: float | None
        sample_to_sample_rms_max: float
        stability_ok: bool

    def run_one_case(reg: Register, mode: str, rate_hz: float,
                     seed: int = SEED) -> tuple[np.ndarray, ModulationResult]:
        """Drive SPU-94 with a sine input + the modulation stream for `reg`;
        returns (stereo_output_int16, ModulationResult)."""
        # Build SPU-94 state; load Hall preset as a known-audible baseline.
        s = SPU94()
        s.load_preset(spu94.Preset.HALL)

        # Input: 440 Hz sine, int16 stereo
        t = np.arange(N) / FS
        input_ch = (AMP * np.sin(2 * np.pi * 440.0 * t)).astype(np.int16)
        Lin = input_ch.copy()
        Rin = input_ch.copy()
        Lout = np.zeros(N, dtype=np.int16)
        Rout = np.zeros(N, dtype=np.int16)

        # Modulation stream in [-1, 1]; map to register's valid range
        norm = modulation_stream(mode, rate_hz or 0.0, N, seed)
        reg_vals = modulation_to_register_values(norm, reg)

        # Drive in 1024-sample blocks; write reg once per block at block-start.
        #
        # LOAD-BEARING BINDING CONTRACT: set_reg_typed needs the raw ctypes
        # state handle, which the Phase 6 SPU94 class exposes as `._state`.
        # We pass `s._state` directly (no hasattr fallback) so a rename in
        # the Phase 6 binding fails LOUDLY with AttributeError rather than
        # silently treating the SPU94 instance as a valid ctypes argument.
        # See threat T-07-04-E below: this attribute is documented as part
        # of SPU94's test-surface contract; breaking it is an API-stability
        # issue to raise in Phase 6 review.
        state_handle = s._state  # AttributeError here = Phase 6 rename regression
        for bs in range(0, N, BLOCK):
            be = min(bs + BLOCK, N)
            set_reg_typed(state_handle, reg, int(reg_vals[bs]))
            s.process(Lin[bs:be], Rin[bs:be], Lout[bs:be], Rout[bs:be])

        # Stability checks
        stereo_out = np.stack([Lout, Rout], axis=1)
        stability_ok = (
            np.all(np.isfinite(stereo_out.astype(np.float64))) and
            int(stereo_out.min()) >= -32768 and
            int(stereo_out.max()) <= 32767
        )

        # Sample-to-sample RMS delta (proxy for zipper audibility)
        diff = np.diff(stereo_out.astype(np.float64), axis=0)
        s2s_rms_max = float(np.sqrt(np.mean(diff ** 2)))

        # Zipper onset proxy: for sine mode, rate at which s2s_rms crosses threshold;
        # for sweep/random_walk we record the max only.
        zipper_onset = rate_hz if (mode == "sine" and
                                    s2s_rms_max > 0.01 * AMP) else None

        import hashlib
        h = hashlib.sha256(stereo_out.tobytes()).hexdigest()

        return stereo_out, ModulationResult(
            reg_name=reg.name, mode=mode, rate_hz=rate_hz,
            out_sha256=h, zipper_onset_hz=zipper_onset,
            sample_to_sample_rms_max=s2s_rms_max, stability_ok=stability_ok,
        )

    def classify_modulation_cost(reg: Register) -> str:
        """Pitfall 7 classifier. HAND column in LEVERS-CATALOG may override."""
        n = reg.name
        if n in IMMEDIATE_REG_NAMES and n.startswith("v"):
            return "free"
        if n.startswith("d"):  # d* are TICK_LATCHED delay pointers
            return "sample-quantized"
        # m* registers (including mBASE) are catastrophic — buffer discontinuity
        return "catastrophic"
    ```

    ```python
    """Phase 7 TEST-05 parametrized stability + determinism gates."""
    import json
    from pathlib import Path
    import numpy as np
    import pytest
    from spu94 import Register
    from modulation_harness import (
        run_one_case, classify_modulation_cost, RATES_HZ,
        ModulationResult,
    )

    REPORT = Path("tests/python/modulation_report.json")

    @pytest.fixture(scope="session")
    def results():
        return {}

    @pytest.mark.parametrize("mode", ["sine", "sweep", "random_walk"])
    @pytest.mark.parametrize("reg", list(Register),
                             ids=[r.name for r in Register])
    def test_stability(reg, mode, results):
        rate = RATES_HZ[4] if mode == "sine" else None  # representative mid-rate
        out, res = run_one_case(reg, mode, rate)
        assert res.stability_ok, (
            f"{reg.name}/{mode}: stability gate failed; "
            f"s2s_rms_max={res.sample_to_sample_rms_max:.3f}"
        )
        results.setdefault(reg.name, {})[mode] = res

    @pytest.mark.parametrize("mode", ["sine", "sweep", "random_walk"])
    @pytest.mark.parametrize("reg", list(Register),
                             ids=[r.name for r in Register])
    def test_determinism(reg, mode):
        rate = RATES_HZ[4] if mode == "sine" else None
        out_a, res_a = run_one_case(reg, mode, rate, seed=0x1094_DADA)
        out_b, res_b = run_one_case(reg, mode, rate, seed=0x1094_DADA)
        assert res_a.out_sha256 == res_b.out_sha256, (
            f"{reg.name}/{mode}: non-deterministic; "
            f"sha_a={res_a.out_sha256[:16]} sha_b={res_b.out_sha256[:16]}"
        )

    def test_report_written(results, request):
        """Runs last in session; writes aggregated report from other tests' results."""
        # Make the dependency explicit — this runs after the parametrized tests.
        # If results is empty, that means the parametrized tests were skipped.
        assert len(results) > 0, "expected parametrized tests to populate results"
        report = {}
        for reg_name, modes in results.items():
            # Find the Register enum entry
            reg = next((r for r in Register if r.name == reg_name), None)
            if reg is None:
                continue
            report[reg_name] = {
                "modulation_cost": classify_modulation_cost(reg),
                "modes": {
                    m: {
                        "stability_ok": res.stability_ok,
                        "sample_to_sample_rms_max": res.sample_to_sample_rms_max,
                        "zipper_onset_hz": res.zipper_onset_hz,
                        "out_sha256": res.out_sha256,
                    } for m, res in modes.items()
                }
            }
        REPORT.write_text(json.dumps(report, indent=2, sort_keys=True))
        assert len(report) == 35, f"expected 35 registers, got {len(report)}"
    ```

    ```python
    #!/usr/bin/env python3
    """D-16: idempotent LEVERS-CATALOG.md AUTO-column writer.

    Reads tests/python/modulation_report.json; writes AUTO columns
    (modulation_cost, zipper_onset) for each of the 35 registers.
    HAND columns (musical_role, m4_grouping) are preserved verbatim.
    Running twice in a row produces a byte-identical file.
    """
    import json
    import re
    import sys
    from pathlib import Path

    CATALOG = Path("docs/LEVERS-CATALOG.md")
    REPORT = Path("tests/python/modulation_report.json")

    # Match data rows only: `| `register` | HAND1 | AUTO1 | AUTO2 | HAND2 |`
    ROW = re.compile(
        r'^\|\s*`(?P<reg>[a-zA-Z0-9_]+)`\s*'
        r'\|(?P<hand1>[^|]*)'
        r'\|(?P<auto1>[^|]*)'
        r'\|(?P<auto2>[^|]*)'
        r'\|(?P<hand2>[^|]*)\|\s*$'
    )

    def format_zipper_onset(modes: dict) -> str:
        # Pick the lowest sine-mode rate where the onset fired
        onsets = [m.get("zipper_onset_hz") for m in modes.values()
                  if m.get("zipper_onset_hz") is not None]
        if not onsets:
            return "clean through 11 kHz"
        lowest = min(onsets)
        return f"~{lowest:.0f} Hz"

    def rewrite_row(line: str, report: dict) -> str:
        m = ROW.match(line)
        if not m:
            return line
        reg = m.group("reg")
        if reg not in report:
            return line  # unknown register; leave untouched
        mod_cost = report[reg]["modulation_cost"]
        zipper = format_zipper_onset(report[reg]["modes"])
        # Rebuild preserving HAND exactly, rewriting AUTO
        return (f"| `{reg}` |{m.group('hand1')}"
                f"| {mod_cost} | {zipper} |{m.group('hand2')}|\n")

    def main() -> int:
        report = json.loads(REPORT.read_text())
        text = CATALOG.read_text()
        out_lines = []
        for line in text.splitlines(keepends=True):
            out_lines.append(rewrite_row(line, report))
        new_text = "".join(out_lines)
        if new_text != text:
            CATALOG.write_text(new_text)
            print(f"Updated {CATALOG} ({len(report)} registers)")
        else:
            print(f"No changes to {CATALOG}")
        return 0

    if __name__ == "__main__":
        sys.exit(main())
    ```
