A minimal JUCE UnitTest runner. Plans 02-04 add `juce::UnitTest` subclasses to this translation unit (or link additional .cpp files into the `standalone_test_runner` target). The runner discovers and runs all registered UnitTests at process start:

```cpp
#include <JuceHeader.h>

// Minimal runner: instantiate all registered juce::UnitTests, run them,
// exit 0 on all-pass, 1 on any failure. Plans 02-04 add test source
// files to the standalone_test_runner CMake target.
int main()
{
    juce::UnitTestRunner runner;
    runner.runAllTests();

    const int failures = runner.getResults().size() > 0
        ? [&]() {
              int f = 0;
              for (int i = 0; i < runner.getResults().size(); ++i)
                  f += runner.getResults().getUnchecked(i)->failures;
              return f;
          }()
        : 0;

    if (failures > 0)
    {
        std::cerr << failures << " test failure(s)." << std::endl;
        return 1;
    }

    std::cout << "All JUCE UnitTests passed." << std::endl;
    return 0;
}
```

pytest fixtures for Phase 8 standalone tests. Plans 02-04 add test modules that import these fixtures. Provides:

```python
"""Phase 8 standalone test fixtures.

Synthetic WAV generators for I/O wrapper round-trip tests (STANDALONE-02),
SHA-256 comparison helpers for golden tests (STANDALONE-03b), and build
directory locators.
"""
import hashlib
import math
import struct
import wave
from pathlib import Path

import pytest


@pytest.fixture
def build_dir():
    """Return the CMake build directory (assumes standard layout)."""
    repo = Path(__file__).resolve().parent.parent.parent
    bd = repo / "build"
    assert bd.exists(), f"Build directory not found: {bd}"
    return bd


def _make_wav(path: Path, sr: int, bd: int, channels: int, duration_s: float) -> Path:
    """Write a synthetic sine-wave WAV at the given spec."""
    n_frames = int(sr * duration_s)
    # 440 Hz sine, amplitude 0.5
    samples = []
    for i in range(n_frames):
        val = 0.5 * math.sin(2.0 * math.pi * 440.0 * i / sr)
        # Scale to bit depth
        if bd == 8:
            # 8-bit WAV is unsigned
            ival = int((val + 1.0) * 127.5)
            ival = max(0, min(255, ival))
            sample_bytes = struct.pack("<B", ival)
        elif bd == 16:
            ival = int(val * 32767)
            ival = max(-32768, min(32767, ival))
            sample_bytes = struct.pack("<h", ival)
        elif bd == 24:
            ival = int(val * 8388607)
            ival = max(-8388608, min(8388607, ival))
            b = ival.to_bytes(3, byteorder="little", signed=True)
            sample_bytes = b
        elif bd == 32:
            ival = int(val * 2147483647)
            ival = max(-2147483648, min(2147483647, ival))
            sample_bytes = struct.pack("<i", ival)
        else:
            raise ValueError(f"Unsupported bit depth: {bd}")
        # Duplicate for stereo
        for _ch in range(channels):
            samples.append(sample_bytes)

    with wave.open(str(path), "wb") as wf:
        wf.setnchannels(channels)
        wf.setsampwidth(bd // 8)
        wf.setframerate(sr)
        wf.writeframes(b"".join(samples))

    return path


@pytest.fixture
def synthetic_wav(tmp_path):
    """Factory fixture: synthetic_wav(sr, bd, channels, duration_s) -> Path."""
    def _factory(sr=44100, bd=16, channels=2, duration_s=0.5):
        name = f"synth_{sr}hz_{bd}bit_{channels}ch.wav"
        return _make_wav(tmp_path / name, sr, bd, channels, duration_s)
    return _factory


def sha256_file(path: Path) -> str:
    """Return hex SHA-256 digest of a file."""
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()
```
